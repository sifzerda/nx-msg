/* -------------------- CONSTANTS -------------------- */

const UUID_PATTERN =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

const EXPIRED_STATUS_CODES = new Set([
  404,
  410,
]);

const DELIVERY_BATCH_SIZE = 10;

/* -------------------- RESPONSE -------------------- */

function json(body, status = 200) {
  return NextResponse.json(body, {
    status,
    headers: {
      "Cache-Control": "no-store",
    },
  });
}

function emptyDeliveryResult() {
  return json({
    success: true,
    sent: 0,
    failed: 0,
    expired: 0,
  });
}

/* -------------------- AUTHENTICATION -------------------- */

function validSecret(request) {
  const authorization =
    request.headers.get("authorization") || "";

  if (!/^Bearer\s+/i.test(authorization)) {
    return false;
  }

  const provided = authorization.replace(
    /^Bearer\s+/i,
    ""
  );

  const expected =
    process.env.PUSH_WEBHOOK_SECRET || "";

  const a = Buffer.from(provided);
  const b = Buffer.from(expected);

  return (
    b.length > 0 &&
    a.length === b.length &&
    timingSafeEqual(a, b)
  );
}

/* -------------------- VALIDATION -------------------- */

function isValidEvent(payload) {
  return (
    payload?.type === "INSERT" &&
    payload?.table === "messages" &&
    payload?.schema === "public"
  );
}

/* -------------------- DATABASE -------------------- */

async function getPushDelivery(admin, messageId) {
  const { data, error } = await admin.rpc(
    "get_message_push_delivery",
    {
      target_message_id: messageId,
    }
  );

  if (error) throw error;

  if (data === null) {
    return null;
  }

  if (
    !data?.message ||
    !Array.isArray(data.subscriptions)
  ) {
    throw new Error(
      "Invalid push delivery response"
    );
  }

  return data;
}

/* -------------------- NOTIFICATION -------------------- */

function createNotification(message, sender) {
  const senderName =
    sender?.display_name?.trim() ||
    sender?.username?.trim() ||
    "Someone";

  return JSON.stringify({
    messageId: message.id,

    title: `${senderName} sent you a message`,

    body: getNotificationPreview(message),

    url: `/messages/${message.conversation_id}`,

    tag: `chat:${message.conversation_id}`,
  });
}

/* -------------------- DELIVERY -------------------- */

async function deliverNotification(
  subscription,
  notification
) {
  if (!isValidPushEndpoint(subscription.endpoint)) {
    return "failed";
  }

  try {
    await sendWebPush(
      subscription,
      notification
    );

    return "sent";
  } catch (error) {
    return EXPIRED_STATUS_CODES.has(
      error?.statusCode
    )
      ? "expired"
      : "failed";
  }
}

/* -------------------- CLEANUP -------------------- */

async function removeExpiredSubscriptions(
  admin,
  subscriptionIds
) {
  if (!subscriptionIds.length) return;

  const { error } = await admin
    .from("push_subscriptions")
    .delete()
    .in("id", subscriptionIds);

  if (error) {
    console.error(
      "[Push] Expired subscription cleanup failed:",
      error.message
    );
  }
}

/* -------------------- BATCHED DELIVERY -------------------- */

async function deliverNotifications(
  admin,
  subscriptions,
  notification
) {
  const counts = {
    sent: 0,
    failed: 0,
    expired: 0,
  };

  const expiredIds = [];

  for (
    let offset = 0;
    offset < subscriptions.length;
    offset += DELIVERY_BATCH_SIZE
  ) {
    const batch = subscriptions.slice(
      offset,
      offset + DELIVERY_BATCH_SIZE
    );

    const results = await Promise.all(
      batch.map((subscription) =>
        deliverNotification(
          subscription,
          notification
        )
      )
    );

    for (
      let index = 0;
      index < results.length;
      index++
    ) {
      const status = results[index];

      counts[status]++;

      if (status === "expired") {
        expiredIds.push(batch[index].id);
      }
    }
  }

  await removeExpiredSubscriptions(
    admin,
    expiredIds
  );

  return counts;
}

/* -------------------- ROUTE -------------------- */

export async function POST(request) {
  if (!validSecret(request)) {
    return json(
      { error: "Unauthorized" },
      401
    );
  }

  try {
    const payload = await request
      .json()
      .catch(() => null);

    if (!isValidEvent(payload)) {
      return json(
        { error: "Invalid event" },
        400
      );
    }

    const messageId = payload.record?.id;

    if (!isUuid(messageId)) {
      return json(
        { error: "Invalid message ID" },
        400
      );
    }

    const admin = createAdminClient();

    const delivery = await getPushDelivery(
      admin,
      messageId
    );

    if (!delivery) {
      return json(
        { error: "Message not found" },
        404
      );
    }

    const {
      message,
      sender,
      subscriptions,
    } = delivery;

    if (!subscriptions.length) {
      return emptyDeliveryResult();
    }

    const notification = createNotification(
      message,
      sender
    );

    const counts = await deliverNotifications(
      admin,
      subscriptions,
      notification
    );

    return json(
      {
        success: counts.failed === 0,
        ...counts,
      },
      counts.failed ? 503 : 200
    );
  } catch (error) {
    console.error(
      "[Push] Delivery failed:",
      {
        message: error?.message,
        code: error?.code,
      }
    );

    return json(
      { error: "Push delivery failed" },
      500
    );
  }
}

import "server-only";
import * as webpush from "web-push";

const TTL = 3600;
const REQUEST_TIMEOUT = 10_000;

const MAX_ENDPOINT_LENGTH = 4096;

const TRUSTED_HOSTS = new Set([
  "fcm.googleapis.com",
  "updates.push.services.mozilla.com",
  "web.push.apple.com",
]);

const TRUSTED_SUFFIXES = [
  ".notify.windows.com",
  ".push.apple.com",
];

let configured = false;

/* -------------------- ENDPOINT VALIDATION -------------------- */

export function isValidPushEndpoint(endpoint) {
  if (
    typeof endpoint !== "string" ||
    !endpoint ||
    endpoint.length > MAX_ENDPOINT_LENGTH ||
    endpoint !== endpoint.trim()
  ) {
    return false;
  }

  try {
    const url = new URL(endpoint);

    if (
      url.protocol !== "https:" ||
      url.username ||
      url.password ||
      url.port ||
      url.hash
    ) {
      return false;
    }

    const hostname = url.hostname.toLowerCase();

    return (
      TRUSTED_HOSTS.has(hostname) ||
      TRUSTED_SUFFIXES.some(
        (suffix) => hostname.endsWith(suffix)
      )
    );
  } catch {
    return false;
  }
}

/* -------------------- CONFIGURATION -------------------- */

function configureWebPush() {
  if (configured) return;

  const subject = process.env.VAPID_SUBJECT;

  const publicKey =
    process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY;

  const privateKey =
    process.env.VAPID_PRIVATE_KEY;

  if (!subject || !publicKey || !privateKey) {
    throw new Error(
      "Missing Web Push environment variables"
    );
  }

  webpush.setVapidDetails(
    subject,
    publicKey,
    privateKey
  );

  configured = true;
}

/* -------------------- DELIVERY -------------------- */

export async function sendWebPush(
  subscription,
  payload
) {
  if (!isValidPushEndpoint(subscription?.endpoint)) {
    throw new Error("Unsupported push endpoint");
  }

  configureWebPush();

  return webpush.sendNotification(
    {
      endpoint: subscription.endpoint,

      keys: {
        p256dh: subscription.p256dh,
        auth: subscription.auth_key,
      },
    },

    typeof payload === "string"
      ? payload
      : JSON.stringify(payload),

    {
      TTL,
      timeout: REQUEST_TIMEOUT,
    }
  );
}

const SERVICE_WORKER_URL = "/sw.js";

const VAPID_PUBLIC_KEY =

const DISABLED_STATE = Object.freeze({

  enabled: false,

const ENABLED_STATE = Object.freeze({
  preference: true,
  enabled: true,

/* -------------------- SUPPORT -------------------- */

export function isPushSupported() {
  return (
    typeof window !== "undefined" &&
    typeof navigator !== "undefined" &&
    window.isSecureContext &&
    "serviceWorker" in navigator &&
    "PushManager" in window &&
    "Notification" in window
  );
}

/* -------------------- VAPID -------------------- */

function getApplicationServerKey() {
  if (!VAPID_PUBLIC_KEY) {
    throw new Error("Missing VAPID public key");
  }

  const padding = "=".repeat(
    (4 - (VAPID_PUBLIC_KEY.length % 4)) % 4
  );

  const base64 = (VAPID_PUBLIC_KEY + padding)
    .replace(/-/g, "+")
    .replace(/_/g, "/");

  return Uint8Array.from(
    atob(base64),
    (char) => char.charCodeAt(0)
  );
}

function matchesApplicationServerKey(
  subscription,
  expectedKey
) {
  const existingKey =
    subscription.options?.applicationServerKey;

  // Some browsers do not expose the existing key.

  if (!existingKey) return true;

  const actualKey = new Uint8Array(existingKey);

  return (
    actualKey.length === expectedKey.length &&
    actualKey.every(
      (byte, index) => byte === expectedKey[index]
    )
  );
}

/* -------------------- API -------------------- */

async function postJSON(url, body) {
  const response = await fetch(url, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    cache: "no-store",
    body: JSON.stringify(body),
  });

  const result = await response
    .json()
    .catch(() => ({}));

  if (!response.ok) {
    throw new Error(
      result.error ||
      `Request failed (${response.status})`
    );
  }

  return result;
}

/* -------------------- PREFERENCE -------------------- */

async function getPushPreference(userId) {
  if (!userId) {
    throw new Error("User ID is required");
  }

  const { data, error } = await createClient()
    .from("profiles")
    .select("notifications_enabled")
    .eq("id", userId)
    .single();

  if (error) throw error;

  return data?.notifications_enabled === true;
}

async function disablePushPreference(userId) {
  if (!userId) {
    throw new Error("User ID is required");
  }

  await postJSON("/api/push/unsubscribe", {
    endpoint: null,
    rememberPreference: true,
  });
}

/* -------------------- SERVICE WORKER -------------------- */

function getRegistration() {
  if (registrationRequest) {
    return registrationRequest;
  }

  registrationRequest = (async () => {
    const serviceWorker = navigator.serviceWorker;

    const existing =
      await serviceWorker.getRegistration("/");

    const worker =
      existing?.active ||
      existing?.waiting ||
      existing?.installing;

    const expectedUrl = new URL(
      SERVICE_WORKER_URL,
      window.location.origin
    ).href;

    if (!existing || worker?.scriptURL !== expectedUrl) {
      await serviceWorker.register(SERVICE_WORKER_URL);
    }

    return serviceWorker.ready;
  })().finally(() => {
    registrationRequest = null;
  });

  return registrationRequest;
}

async function getExistingSubscription() {
  const registration =
    await navigator.serviceWorker.getRegistration("/");

  return registration
    ? registration.pushManager.getSubscription()
    : null;
}

/* -------------------- SUBSCRIPTION -------------------- */

async function getOrCreateSubscription() {
  const applicationServerKey =
    getApplicationServerKey();

  const registration = await getRegistration();

  const { pushManager } = registration;

  const existing =
    await pushManager.getSubscription();

  if (existing) {
    if (
      !matchesApplicationServerKey(
        existing,
        applicationServerKey
      )
    ) {
      throw new Error(
        "The existing push subscription uses a different VAPID key. Disable notifications and enable them again."
      );
    }

    return existing;
  }

  return pushManager.subscribe({
    userVisibleOnly: true,
    applicationServerKey,
  });
}

async function registerSubscription(
  subscription,
  rememberPreference
) {
  await postJSON("/api/push/subscribe", {
    subscription: subscription.toJSON(),
    rememberPreference,
  });

  return ENABLED_STATE;
}

/* -------------------- RESTORE -------------------- */

export async function restorePushNotifications(userId) {
  if (!isPushSupported()) {
    return DISABLED_STATE;
  }

  const preference =
    await getPushPreference(userId);

  if (!preference) {
    return DISABLED_STATE;
  }

  if (Notification.permission !== "granted") {
    return {
      preference: true,
      enabled: false,
    };
  }

  const subscription =
    await getOrCreateSubscription();

  return registerSubscription(subscription, false);
}

/* -------------------- ENABLE -------------------- */

export async function enablePushNotifications() {
  if (!isPushSupported()) {
    throw new Error(
      "Push notifications are not supported"
    );
  }

  if (Notification.permission === "denied") {
    throw new Error(
      "Notifications are blocked by your browser"
    );
  }

  const permission =
    Notification.permission === "granted"
      ? "granted"
      : await Notification.requestPermission();

  if (permission !== "granted") {
    throw new Error(
      "Notification permission was not granted"
    );
  }

  const subscription =
    await getOrCreateSubscription();

  return registerSubscription(subscription, true);
}

/* -------------------- SUBSCRIPTION REMOVAL -------------------- */

async function removeSubscription(
  subscription,
  {
    rememberPreference,
    removeOnServerFailure = false,
  }
) {
  let error = null;

  try {
    await postJSON("/api/push/unsubscribe", {
      endpoint: subscription.endpoint,
      rememberPreference,
    });
  } catch (cause) {
    if (!removeOnServerFailure) {
      throw cause;
    }

    error = cause;
  }

  let warning = null;

  try {
    const removed =
      await subscription.unsubscribe();

    if (!removed) {
      warning =
        "Browser subscription could not be removed.";
    }
  } catch {
    warning =
      "Browser subscription cleanup failed.";
  }

  return {
    error,
    warning,
  };
}

/* -------------------- DISABLE -------------------- */

export async function disablePushNotifications(userId) {
  const subscription =
    await getExistingSubscription();

  if (!subscription) {
    await disablePushPreference(userId);

    return DISABLED_STATE;
  }

  const { warning } = await removeSubscription(
    subscription,
    {
      rememberPreference: true,
    }
  );

  return {
    ...DISABLED_STATE,
    warning,
  };
}

/* -------------------- DISMISS NOTIFICATIONS -------------------- */

async function dismissNotifications() {
  const registration =
    await navigator.serviceWorker.getRegistration("/");

  if (!registration) return;

  const notifications =
    await registration.getNotifications();

  for (const notification of notifications) {
    if (
      notification.tag === "messenger" ||
      notification.tag.startsWith("chat:")
    ) {
      notification.close();
    }
  }
}

/* -------------------- LOGOUT CLEANUP -------------------- */

export async function cleanupPushForLogout() {
  if (
    typeof navigator === "undefined" ||
    !("serviceWorker" in navigator)
  ) {
    return null;
  }

  let cleanupError = null;

  try {
    const subscription =
      await getExistingSubscription();

    if (subscription) {
      const { error, warning } =
        await removeSubscription(subscription, {
          rememberPreference: false,
          removeOnServerFailure: true,
        });

      cleanupError =
        error ||
        (warning ? new Error(warning) : null);
    }
  } catch (error) {
    cleanupError = error;
  }

  try {
    await dismissNotifications();
  } catch (error) {
    cleanupError ??= error;
  }

  return cleanupError;
}

import { create } from "zustand";

import {
  isPushSupported,
  restorePushNotifications,
  enablePushNotifications,
  disablePushNotifications,
} from "@/lib/pushClient";

const INITIAL_STATE = {
  userId: null,

  supported: false,
  initialized: false,
  checking: false,
  loading: false,

  preference: false,
  enabled: false,

  error: null,
  warning: null,

  signingOut: false,
};

let generation = 0;
let activeRequest = null;

const usePushStore = create((set, get) => {
  function runOperation(
    operation,
    {
      fallback,
      busyKey = "loading",
      initialState = {},
    }
  ) {
    const current = generation;

    let resolve;
    let reject;

    const result = new Promise((res, rej) => {
      resolve = res;
      reject = rej;
    });

    const request = result
      .then((value) => {
        if (current !== generation) {
          return false;
        }

        set({
          initialized: true,
          enabled: value.enabled === true,
          preference: value.preference === true,
          warning: value.warning || null,
        });

        return true;
      })
      .catch((error) => {
        if (current === generation) {
          set({
            error: error?.message || fallback,
          });
        }

        return false;
      })
      .finally(() => {
        if (activeRequest === request) {
          activeRequest = null;
        }

        if (current === generation) {
          set({ [busyKey]: false });
        }
      });

    activeRequest = request;

    set({
      ...initialState,
      [busyKey]: true,
      error: null,
      warning: null,
    });

    try {
      if (
        current !== generation ||
        get().signingOut
      ) {
        throw new Error(
          "Notification operation cancelled"
        );
      }

      // Preserve browser user activation.
      resolve(operation());
    } catch (error) {
      reject(error);
    }

    return request;
  }

  function canMutate() {
    const state = get();

    return (
      Boolean(state.userId) &&
      state.supported &&
      !state.checking &&
      !state.loading &&
      !state.signingOut &&
      !activeRequest
    );
  }

  function initialize(userId, { force = false } = {}) {
    const state = get();

    if (!userId || state.signingOut) {
      return Promise.resolve(false);
    }

    if (
      state.userId &&
      state.userId !== userId
    ) {
      return Promise.resolve(false);
    }

    if (activeRequest) {
      return state.userId === userId
        ? activeRequest
        : Promise.resolve(false);
    }

    const sameUser = state.userId === userId;

    if (
      sameUser &&
      state.initialized &&
      !force
    ) {
      return Promise.resolve(true);
    }

    generation++;

    const supported = isPushSupported();

    if (!supported) {
      set({
        ...INITIAL_STATE,
        userId,
        initialized: true,
      });

      return Promise.resolve(true);
    }

    return runOperation(
      () => restorePushNotifications(userId),
      {
        fallback: "Failed to restore notifications",
        busyKey: "checking",

        initialState: {
          ...(sameUser ? {} : INITIAL_STATE),
          userId,
          supported: true,
        },
      }
    );
  }

  function enable() {
    if (!canMutate()) {
      return Promise.resolve(false);
    }

    return runOperation(
      enablePushNotifications,
      {
        fallback: "Failed to enable notifications",
      }
    );
  }

  function disable() {
    if (!canMutate()) {
      return Promise.resolve(false);
    }

    const { userId } = get();

    return runOperation(
      () => disablePushNotifications(userId),
      {
        fallback: "Failed to disable notifications",
      }
    );
  }

  async function prepareForLogout() {
    set({ signingOut: true });

    if (activeRequest) {
      await activeRequest;
    }
  }

  return {
    ...INITIAL_STATE,

    initialize,
    enable,
    disable,
    prepareForLogout,

    refresh: () =>
      initialize(get().userId, { force: true }),

    clearError: () =>
      set({
        error: null,
        warning: null,
      }),

    cancelLogout: () =>
      set({ signingOut: false }),

    reset: () => {
      generation++;
      set({ ...INITIAL_STATE });
    },
  };
});

export default usePushStore;

/* -------------------- MESSAGE HELPERS -------------------- */

function sameItems(a, b) {
  return (
    a.length === b.length &&
    a.every((item, index) => item === b[index])
  );
}

function compareMessages(a, b) {
  return (
    Date.parse(a.created_at) - Date.parse(b.created_at) ||
    a.id.localeCompare(b.id)
  );
}

function mergeMessages(existing, incoming) {
  const unique = new Map(
    incoming.map((message) => [message.id, message])
  );

  // Existing Realtime data takes precedence.
  for (const message of existing) {
    unique.set(message.id, message);
  }

  const next = [...unique.values()].sort(compareMessages);

  return sameItems(existing, next) ? existing : next;
}

function insertMessage(messages, message) {
  if (messages.some((item) => item.id === message.id)) {
    return messages;
  }

  const last = messages.at(-1);

  // Fast path for chronological messages.
  if (!last || compareMessages(last, message) <= 0) {
    return [...messages, message];
  }

  // Binary search for out-of-order messages.
  let low = 0;
  let high = messages.length;

  while (low < high) {
    const middle = (low + high) >>> 1;

    if (compareMessages(messages[middle], message) <= 0) {
      low = middle + 1;
    } else {
      high = middle;
    }
  }

  const next = [...messages];

  next.splice(low, 0, message);

  return next;
}

/* -------------------- CONVERSATION HELPERS -------------------- */

function reconcileConversations(previous, incoming) {
  const byId = new Map(
    previous.map((chat) => [chat.id, chat])
  );

  const next = incoming.map((chat) => {
    const existing = byId.get(chat.id);

    return existing && shallow(existing, chat)
      ? existing
      : chat;
  });

  return sameItems(previous, next) ? previous : next;
}

const useChatStore = create((set, get) => {

  /* -------------------- INTERNAL HELPERS -------------------- */

  function createChatGuard(chatId) {
    const currentGeneration = generation;
    const requestId = chatRequestId;

    return () =>
      currentGeneration === generation &&
      requestId === chatRequestId &&
      get().activeChatId === chatId;

  function updateMessages(incoming) {

    if (messages !== current) {

    }
  }

  function setError(error, fallback) {

  /* -------------------- CONVERSATIONS -------------------- */

  function fetchConversations({ force = false } = {}) {
    if (conversationRequest) {
      if (force) {
        refreshPending = true;
      }

      return conversationRequest;
    }

    const currentGeneration = generation;

    const request = Promise.resolve()
      .then(async () => {
        let success = false;

        while (currentGeneration === generation) {
          refreshPending = false;

          try {
            const incoming = await getConversations();

            if (currentGeneration !== generation) {
              return false;
            }

            // A mutation occurred during the request.
            // Retrieve a fresh authoritative snapshot.
            if (refreshPending) continue;

            const current = get().conversations;

            const conversations = reconcileConversations(
              current,
              incoming
            );

            if (conversations !== current) {
              set({ conversations });
            }

            success = true;
          } catch (error) {
            if (currentGeneration !== generation) {
              return false;
            }

            success = false;

            setError(
              error,
              "Failed to refresh conversations"
            );
          }

          if (!refreshPending) break;
        }

        return (
          success &&
          currentGeneration === generation
        );
      })
      .finally(() => {
        if (conversationRequest === request) {
          conversationRequest = null;
        }
      });

    conversationRequest = request;

    return request;
  }

  /* -------------------- OPEN CHAT -------------------- */

  async function openChat(chatId) {
    chatRequestId++;

    const isCurrent = createChatGuard(chatId);

    set({
      ...CHAT_STATE,
      activeChatId: chatId,
      loading: true,
    });

    try {
      const messages = await getMessages(chatId);

      if (!isCurrent()) return;

      set((state) => ({
        messages: mergeMessages(
          state.messages,
          messages
        ),
        loading: false,
      }));

      if (isPageVisible()) {
        await get().markRead(chatId);
      }
    } catch (error) {
      if (isCurrent()) {
        setError(error, "Failed to load messages");
      }
    } finally {
      if (isCurrent() && get().loading) {
        set({ loading: false });
      }
    }
  }

  /* -------------------- SYNCHRONISE MESSAGES -------------------- */

  /* -------------------- SYNCHRONISE MESSAGES -------------------- */

function syncMessages() {
  const chatId = get().activeChatId;

  if (!chatId) {
    return Promise.resolve(false);
  }

  const currentRequestId = chatRequestId;

  if (
    messageSyncRequest?.chatId === chatId &&
    messageSyncRequest?.requestId === currentRequestId
  ) {
    return messageSyncRequest.promise;
  }

  const isCurrent = createChatGuard(chatId);

  const entry = {
    chatId,
    requestId: currentRequestId,
    promise: null,
  };

  entry.promise = Promise.resolve()
    .then(async () => {
      if (!isCurrent()) return false;

      const incoming = await getMessages(chatId);

      if (!isCurrent()) return false;

      updateMessages(incoming);

      return true;
    })
    .catch((error) => {
      if (isCurrent()) {
        console.error(
          "Failed to synchronise messages:",
          error
        );
      }

      return false;
    })
    .finally(() => {
      if (messageSyncRequest === entry) {
        messageSyncRequest = null;
      }
    });

  messageSyncRequest = entry;

  return entry.promise;
}

  /* -------------------- RECEIVE MESSAGE -------------------- */

  /* -------------------- RECEIVE MESSAGE -------------------- */

function receiveMessage(message) {
  const state = get();

  if (
    !message?.id ||
    message.conversation_id !== state.activeChatId
  ) {
    return false;
  }

  const messages = insertMessage(
    state.messages,
    message
  );

  if (messages === state.messages) {
    return false;
  }

  set({ messages });

  return true;
}

  /* -------------------- READ RECEIPTS -------------------- */

function markRead(chatId) {
  if (!chatId) {
    return Promise.resolve(false);
  }

  const existing = readRequests.get(chatId);

  if (existing) {
    existing.pending = true;
    return existing.promise;
  }

  const currentGeneration = generation;

  const entry = {
    pending: false,
    promise: null,
  };

  entry.promise = Promise.resolve()
    .then(async () => {
      while (currentGeneration === generation) {
        entry.pending = false;

        await markChatRead(chatId);

        if (currentGeneration !== generation) {
          return false;
        }

        if (entry.pending) continue;

        const refreshed =
          await get().fetchConversations({
            force: true,
          });

        if (currentGeneration !== generation) {
          return false;
        }

        if (!entry.pending) {
          return refreshed === true;
        }
      }

      return false;
    })
    .catch((error) => {
      if (currentGeneration === generation) {
        setError(
          error,
          "Failed to mark messages as read"
        );
      }

      return false;
    })
    .finally(() => {
      if (readRequests.get(chatId) === entry) {
        readRequests.delete(chatId);
      }
    });

  readRequests.set(chatId, entry);

  return entry.promise;
}

  /* -------------------- SEND MESSAGE -------------------- */

  async function send(
    content,
    attachmentFile = null,
    userId = null,
    gifId = null
  ) {
    const state = get();

    const {
      activeChatId: chatId,
      sending,
      replyTo,
    } = state;

    if (!chatId || sending) {
      return false;
    }

    if (
      typeof content === "string" &&
      !content.trim() &&
      !attachmentFile &&
      !gifId
    ) {
      return false;
    }

    const unavailable = "Messaging is unavailable in this conversation";

    let message;

    // Validate before uploading any attachments.
    try {
      if (!canSendMessage(state, chatId)) {
        throw new Error(unavailable);
      }

      message = validateMessage(
        content,
        attachmentFile,
        gifId
      );

      if (attachmentFile && !userId) {
        throw new Error("Missing user information");
      }
    } catch (error) {
      setError(error, "Invalid message");

      return false;
    }

    const currentGeneration = generation;

    const replyToId =
      replyTo?.conversation_id === chatId
        ? replyTo.id
        : null;

    set({
      sending: true,
      error: null,
    });

    try {
      const attachment = attachmentFile
        ? await uploadAttachment({
          conversationId: chatId,
          userId,
          file: attachmentFile,
        })
        : null;

      // Ignore operations belonging to a previous session.
      if (currentGeneration !== generation) {
        return false;
      }

      // Permissions may have changed during the upload.
      if (!canSendMessage(get(), chatId)) {
        throw new Error(unavailable);
      }

      const result = await sendMessage(
        chatId,
        message,
        replyToId,
        attachment,
        gifId
      );

      if (currentGeneration !== generation) {
        return false;
      }

      get().receiveMessage(result);

      const current = get();

      // Do not clear a newly selected reply.
      if (
        current.activeChatId === chatId &&
        current.replyTo === replyTo &&
        current.replyTo !== null
      ) {
        set({ replyTo: null });
      }

      void get().fetchConversations({ force: true });

      return true;
    } catch (error) {
      if (currentGeneration === generation) {
        if (get().activeChatId === chatId) {
          setError(error, "Failed to send message");
        }

        // Refresh permissions after an authorization failure.
        if (error?.code === "42501") {
          void get().fetchConversations({ force: true });
        }
      }

      return false;
    } finally {
      if (currentGeneration === generation) {
        set({ sending: false });
      }
    }
  }

  /* -------------------- CLOSE CHAT -------------------- */

  function closeChat(chatId) {
    if (get().activeChatId !== chatId) return;

    chatRequestId++;

    set({ ...CHAT_STATE });
  }

  /* -------------------- REPLIES -------------------- */

  function setReplyTo(message) {
    if (
      message &&
      message.conversation_id !== get().activeChatId
    ) {
      return;
    }

    const replyTo = message ?? null;

    if (get().replyTo !== replyTo) {
      set({ replyTo });
    }
  }

  function clearReply() {
    if (get().replyTo !== null) {
      set({ replyTo: null });
    }
  }

  /* -------------------- RESET -------------------- */

function reset() {
  generation++;
  chatRequestId++;

  conversationRequest = null;
  messageSyncRequest = null;

  refreshPending = false;

  readRequests.clear();

  set({ ...INITIAL_STATE });
}

  /* -------------------- PUBLIC INTERFACE -------------------- */

  return {
    ...INITIAL_STATE,

    fetchConversations,
    openChat,
    closeChat,
    syncMessages,
    receiveMessage,
    markRead,
    send,
    setReplyTo,
    clearReply,
    reset,
  };
});

const DEFAULT_PATH = "/messages";
const UUID_SOURCE = "[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}";

const UUID_PATTERN = new RegExp(`^${UUID_SOURCE}$`, "i");
const CHAT_PATH_PATTERN = new RegExp(`^/messages/(${UUID_SOURCE})$`, "i");

const MAX_TITLE_LENGTH = 120;
const MAX_BODY_LENGTH = 250;

/* -------------------- VALIDATION -------------------- */

function getSafePath(value) {
  return (
    typeof value === "string" &&
    value.length === DEFAULT_PATH.length + 37 &&
    CHAT_PATH_PATTERN.test(value)
  )
    ? value
    : DEFAULT_PATH;
}

function getMessageId(value) {
  return (
    typeof value === "string" &&
    value.length === 36 &&
    UUID_PATTERN.test(value)
  )
    ? value.toLowerCase()
    : null;
}

function getText(value, fallback, maxLength) {
  const text =
    typeof value === "string"
      ? value.trim()
      : "";

  return (text || fallback).slice(0, maxLength);
}

function getNotificationTag(path) {
  const match = CHAT_PATH_PATTERN.exec(path);

  return match
    ? `chat:${match[1].toLowerCase()}`
    : "messenger";
}

function parsePushData(event) {
  try {
    const data = event.data?.json();

    return (
      data &&
      typeof data === "object" &&
      !Array.isArray(data)
    )
      ? data
      : {};
  } catch {
    return {};
  }
}

/* -------------------- PUSH -------------------- */

async function handlePush(event) {
  const data = parsePushData(event);

  const path = getSafePath(data.url);
  const messageId = getMessageId(data.messageId);
  const tag = getNotificationTag(path);

  const title = getText(
    data.title,
    "New message",
    MAX_TITLE_LENGTH
  );

  const body = getText(
    data.body,
    "You have a new message",
    MAX_BODY_LENGTH
  );

  let duplicate = false;

  if (messageId) {
    try {
      const existing =
        await self.registration.getNotifications({
          tag,
        });

      duplicate = existing.some(
        (notification) =>
          notification.data?.messageId === messageId
      );
    } catch {
      // Continue displaying notifications if
      // the browser cannot inspect existing ones.
    }
  }

  await self.registration.showNotification(title, {
    body,

    icon: "/favicon.ico",

    tag,

    renotify: !duplicate,

    silent: duplicate,

    data: {
      url: path,
      messageId,
    },
  });
}

self.addEventListener("push", (event) => {
  event.waitUntil(handlePush(event));
});

/* -------------------- WINDOW NAVIGATION -------------------- */

async function focusMessenger(path) {
  const origin = self.location.origin;

  const targetUrl = new URL(path, origin);

  const windows = await self.clients.matchAll({
    type: "window",
    includeUncontrolled: true,
  });

  let reusableWindow = null;

  for (const client of windows) {
    const url = new URL(client.url);

    if (url.origin !== origin) {
      continue;
    }

    if (url.pathname === path) {
      try {
        return await client.focus();
      } catch {
        continue;
      }
    }

    const isMessenger =
      url.pathname === DEFAULT_PATH ||
      url.pathname.startsWith("/messages/");

    if (isMessenger && !reusableWindow) {
      reusableWindow = client;
    }
  }

  if (reusableWindow) {
    try {
      const navigated =
        await reusableWindow.navigate(targetUrl.href);

      if (navigated) {
        return await navigated.focus();
      }
    } catch {
      // Open a new window if reuse fails.
    }
  }

  return self.clients.openWindow(targetUrl.href);
}

/* -------------------- NOTIFICATION CLICK -------------------- */

self.addEventListener("notificationclick", (event) => {
  event.notification.close();

  const path = getSafePath(
    event.notification.data?.url
  );

  event.waitUntil(focusMessenger(path));
});
