/* -------------------- HELPERS -------------------- */

function getProfileName(profile) {
  return (
    profile?.display_name ||
    profile?.username ||
    "Member"
  );
}

function getChatName(chat) {
  return chat?.type === "group"
    ? chat.group_name || "Unnamed Group"
    : chat?.other_display_name ||
    chat?.other_username ||
    "Conversation";
}

function scrollToBottom(container) {
  if (container) {
    container.scrollTop = container.scrollHeight;
  }
}

function getTypingLabel(names) {
  const count = names.length;

  if (!count) return "";

  if (count === 1) {
    return `${names[0]} is typing...`;
  }

  if (count <= 3) {
    return `${names.slice(0, -1).join(", ")} and ${names[count - 1]
      } are typing...`;
  }

  return `${count} people are typing...`;
}

function setReplyTo(message) {
  useChatStore.getState().setReplyTo(message);
}

/* -------------------- GROUP PROFILES -------------------- */

function useGroupProfiles({
  isGroup,
  messages,
  replyMessages,
  typingUsers,
  userId,
  memberProfiles,
  membershipLoading,
}) {
  const [historicalProfiles, setHistoricalProfiles] = useState(EMPTY_PROFILES);

  const senderKey = useMemo(() => {
    if (!isGroup || membershipLoading) return "";

    const ids = new Set();

    function addId(id) {
      if (
        id &&
        id !== userId &&
        !memberProfiles[id]
      ) {
        ids.add(id);
      }
    }

    for (const message of messages) {
      addId(message.sender_id);
    }

    for (const message of replyMessages.values()) {
      addId(message?.sender_id);
    }

    for (const id of typingUsers) {
      addId(id);
    }

    return [...ids].sort().join(",");
  }, [
    isGroup,
    messages,
    replyMessages,
    typingUsers,
    userId,
    memberProfiles,
  ]);

  useEffect(() => {
    if (!senderKey) return;

    let active = true;

    async function loadProfiles() {
      try {
        const result = await getMessageProfiles(
          senderKey.split(",")
        );

        if (!active || !result) return;

        setHistoricalProfiles((current) => {
          let next = current;

          for (const [id, profile] of Object.entries(result)) {
            const previous = current[id];

            if (
              previous &&
              previous.display_name === profile?.display_name &&
              previous.username === profile?.username
            ) {
              continue;
            }

            if (next === current) {
              next = { ...current };
            }

            next[id] = profile;
          }

          return next;
        });
      } catch (error) {
        if (active) {
          console.error(
            "Failed to load sender profiles:",
            error
          );
        }
      }
    }

    void loadProfiles();

    return () => {
      active = false;
    };
  }, [senderKey]);

  return useMemo(
    () => ({
      ...historicalProfiles,
      ...memberProfiles,
    }),
    [historicalProfiles, memberProfiles]
  );
}

/* -------------------- MESSAGE ROW -------------------- */

const MessageRow = memo(function MessageRow({
  message,
  userId,
  isGroup,
  isRead,
  senderName,
  quotedMessage,
  quotedSenderName,

  return (

      className="rounded-xl"
    >
      {isGroup && !isOwn && (

      )}

        message={message}
        isOwn={isOwn}
        isRead={isRead}
        showReadStatus={!isGroup}
        quotedMessage={quotedMessage}
        quotedSenderName={quotedSenderName}
        onReply={onReply}
        onJumpToMessage={onJumpToMessage}
        userId={userId}
        reactions={reactions}
        onToggleReaction={onToggleReaction}
      />
    </div>
  );
});

/* -------------------- MESSAGE LIST -------------------- */

const MessageList = memo(function MessageList({
  messages,
  loading,
  userId,
  isGroup,
  canSendMessages,
  recipientReadTime,
  replyMessages,
  reactionsByMessage,
  getSenderName,
  onJumpToMessage,
  onToggleReaction,
  containerRef,
  onScroll,
}) {
  const now = useCurrentDay();

  const groups = useMemo(
    () => groupMessagesByDate(messages),
    [messages]
  );

  const onReply = canSendMessages
    ? setReplyTo
    : undefined;

  return (
    <div
      ref={containerRef}
      onScroll={onScroll}
      className="min-h-0 min-w-0 overflow-y-auto overscroll-contain p-4"
    >
      <div className="space-y-3">
        {loading && (
          <p className="text-gray-400">
            Loading messages...
          </p>
        )}

        {!loading && !messages.length && (
          <p className="text-center text-gray-500">
            {canSendMessages
              ? "No messages yet. Say hello!"
              : "No messages yet."}
          </p>
        )}

        {groups.map((group) => (
          <section
            key={group.key}
            className="space-y-3"
          >
            <DateSeparator
              date={group.date}
              now={now}
            />

            {group.messages.map((message) => {
              const isOwn = message.sender_id === userId;

              const quotedMessage = replyMessages.get(
                message.reply_to_id
              );

              const isRead =
                !isGroup &&
                isOwn &&
                recipientReadTime > 0 &&
                recipientReadTime >= Date.parse(message.created_at);

              return (
                <MessageRow
                  key={message.id}
                  message={message}
                  userId={userId}
                  isGroup={isGroup}
                  isRead={isRead}
                  senderName={
                    isGroup && !isOwn
                      ? getSenderName(message)
                      : undefined
                  }
                  quotedMessage={quotedMessage}
                  quotedSenderName={
                    message.reply_to_id
                      ? getSenderName(quotedMessage)
                      : undefined
                  }
                  reactions={
                    reactionsByMessage.get(message.id) ??
                    EMPTY_ARRAY
                  }
                  onReply={onReply}
                  onJumpToMessage={onJumpToMessage}
                  onToggleReaction={onToggleReaction}
                />
              );
            })}
          </section>
        ))}
      </div>
    </div>
  );
});

/* -------------------- CONVERSATION HEADER -------------------- */

const ConversationHeader = memo(function ConversationHeader({
  conversationId,
  userId,
  chat,
  chatName,
  isGroup,
  isPrivate,
  isBlockedByMe,
  messagingUnavailable,
  messages,
  onJumpToMessage,
}) {
  const [searchOpen, setSearchOpen] = useState(false);
  const [exporting, setExporting] = useState(false);
  const [exportError, setExportError] = useState(null);

  const exportingRef = useRef(false);

  const closeSearch = useCallback(() => {
    setSearchOpen(false);
  }, []);

  async function handleExportChat() {
    if (!conversationId || exportingRef.current) return;

    exportingRef.current = true;

    setExporting(true);
    setExportError(null);

    try {
      const { exportChatLog } = await import(
        "@/lib/chatExport"
      );

      await exportChatLog({
        id: conversationId,
        name: chatName,
        type: chat?.type,
      });
    } catch (error) {
      console.error("Failed to export chat:", error);

      setExportError(
        error?.message || "Unable to download chat log."
      );
    } finally {
      exportingRef.current = false;
      setExporting(false);
    }
  }

  return (
    <div className="min-w-0">
      <header className="flex items-center gap-3 border-b border-gray-800 p-4">
        {isGroup && (
          <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-blue-600/20 text-blue-400">
            <UsersRound size={22} />
          </div>
        )}

        <div className="min-w-0 flex-1">
          <h2 className="truncate text-lg font-semibold text-white">
            {chatName}
          </h2>

          {isGroup && (
            <p className="text-xs text-gray-400">
              {chat?.member_count ?? 0} members
            </p>
          )}

          {isPrivate && messagingUnavailable && (
            <p className="mt-0.5 text-xs text-gray-400">
              {isBlockedByMe
                ? "You have blocked this user"
                : "Messaging unavailable"}
            </p>
          )}
        </div>

        <button
          type="button"
          onClick={() => setSearchOpen((value) => !value)}
          aria-label="Search conversation"
          aria-expanded={searchOpen}
          title="Search conversation"
          className={`flex shrink-0 items-center justify-center rounded-lg border p-2.5 transition-colors ${searchOpen
            ? "border-blue-500 bg-blue-600/20 text-blue-400"
            : "border-gray-700 bg-gray-900 text-gray-300 hover:border-blue-500 hover:text-white"
            }`}
        >
          <Search size={18} />
        </button>

        <button
          type="button"
          onClick={handleExportChat}
          disabled={!conversationId || exporting}
          title="Download complete chat history"
          aria-label="Download chat log"
          className="flex shrink-0 items-center gap-2 rounded-lg border border-gray-700 bg-gray-900 px-3 py-2 text-sm font-medium text-gray-200 transition-colors hover:border-blue-500 hover:bg-gray-800 hover:text-white disabled:cursor-not-allowed disabled:opacity-50"
        >
          <Download size={17} />

          <span className="hidden sm:inline">
            {exporting ? "Exporting..." : "Download chat"}
          </span>
        </button>

        {isPrivate && chat?.other_user_id && (
          <ChatBlockMenu
            key={`${userId}:${conversationId}`}
            userId={chat.other_user_id}
            displayName={chatName}
            isBlocked={isBlockedByMe}
          />
        )}
      </header>

      {exportError && (
        <div
          role="alert"
          className="border-b border-red-900/40 bg-red-950/30 px-4 py-2 text-sm text-red-400"
        >
          {exportError}
        </div>
      )}

      {searchOpen && (
        <ConversationSearch
          conversationId={conversationId}
          userId={userId}
          messages={messages}
          onClose={closeSearch}
          onJumpToMessage={onJumpToMessage}
        />
      )}
    </div>
  );
});

/* -------------------- CHAT SESSION -------------------- */

function ChatSession({
  conversationId,
  userId,
}) {
  const messagesContainerRef = useRef(null);
  const shouldScrollRef = useRef(true);
  const previousMessageIdRef = useRef(null);

  const {
    messages,
    chat,
    loading,
    error,
    replyTo,
  } = useChatStore(
    useShallow((state) => ({
      messages:
        state.activeChatId === conversationId
          ? state.messages
          : EMPTY_ARRAY,

      chat: state.conversations.find(
        (item) => item.id === conversationId
      ),

      loading:
        state.activeChatId === conversationId
          ? state.loading
          : false,

      error:
        state.activeChatId === conversationId
          ? state.error
          : null,

      replyTo:
        state.activeChatId === conversationId
          ? state.replyTo
          : null,
    }))
  );

  const isGroup = chat?.type === "group";
  const isPrivate = chat?.type === "private";

  const isBlockedByMe =
    isPrivate && chat?.is_blocked_by_me === true;

  const canSendMessages =
    chat?.can_send_messages === true;

  const messagingUnavailable =
    chat?.can_send_messages === false;

  const chatName = getChatName(chat);

  /* -------------------- MESSAGE DATA -------------------- */

  const replyMessages = useReplyMessages(
    conversationId,
    messages
  );

  const {
    reactionsByMessage,
    toggleReaction,
    reactionError,
  } = useMessageReactions(
    conversationId,
    userId,
    messages
  );

  const {
    otherMembers,
    profiles: memberProfiles,
    recipientReadTime,
    error: membershipError,
  } = useConversationMembers(
    conversationId,
    userId,
    isPrivate || isGroup
  );

  const {
    typingUsers,
    notifyTyping,
    stopTyping,
  } = useTypingIndicator(
    conversationId,
    userId,
    { enabled: canSendMessages }
  );

  const profiles = useGroupProfiles({
    isGroup,
    messages,
    replyMessages,
    typingUsers,
    userId,
    memberProfiles,
  });

  /* -------------------- SENDER NAMES -------------------- */

  const getSenderName = useCallback(
    (message) => {
      if (!message) return "Member";

      if (message.sender_id === userId) {
        return "You";
      }

      return isGroup
        ? getProfileName(profiles[message.sender_id])
        : chatName;
    },
    [userId, isGroup, profiles, chatName]
  );

  const otherUserId = chat?.other_user_id;

  const typingNames = useMemo(() => {
    if (!typingUsers.length) return EMPTY_ARRAY;

    if (!isGroup) {
      return otherUserId &&
        typingUsers.includes(otherUserId)
        ? [chatName]
        : EMPTY_ARRAY;
    }

    return typingUsers.map((id) =>
      getProfileName(profiles[id])
    );
  }, [
    isGroup,
    otherUserId,
    chatName,
    typingUsers,
    profiles,
  ]);

  /* -------------------- CHAT LIFECYCLE -------------------- */

  useEffect(() => {
    if (!conversationId) return;

    void useChatStore.getState().openChat(conversationId);

    return () => {
      useChatStore.getState().closeChat(conversationId);
    };
  }, [conversationId]);

  useEffect(() => {
    if (!messagingUnavailable) return;

    stopTyping();
    useChatStore.getState().clearReply();
  }, [messagingUnavailable, stopTyping]);

  /* -------------------- SCROLL MANAGEMENT -------------------- */

  const handleScroll = useCallback(() => {
    const container = messagesContainerRef.current;

    if (!container) return;

    shouldScrollRef.current =
      container.scrollHeight -
      container.scrollTop -
      container.clientHeight <=
      SCROLL_THRESHOLD;
  }, []);

  useLayoutEffect(() => {
    const container = messagesContainerRef.current;
    const latestMessage = messages.at(-1);

    if (!container) return;

    if (!latestMessage) {
      previousMessageIdRef.current = null;
      shouldScrollRef.current = true;
      return;
    }

    const previousId = previousMessageIdRef.current;

    const initialLoad = previousId === null;

    const ownNewMessage =
      previousId !== latestMessage.id &&
      latestMessage.sender_id === userId;

    if (
      initialLoad ||
      ownNewMessage ||
      shouldScrollRef.current
    ) {
      scrollToBottom(container);
      shouldScrollRef.current = true;
    }

    previousMessageIdRef.current = latestMessage.id;
  }, [messages, userId]);

  useEffect(() => {
    const container = messagesContainerRef.current;

    if (
      !container ||
      typeof ResizeObserver === "undefined"
    ) {
      return;
    }

    const observer = new ResizeObserver(() => {
      if (shouldScrollRef.current) {
        scrollToBottom(container);
      }
    });

    observer.observe(container);

    // Also observe message content so loaded images,
    // GIFs and previews can preserve bottom alignment.

    if (container.firstElementChild) {
      observer.observe(container.firstElementChild);
    }

    return () => observer.disconnect();
  }, []);

  const jumpToMessage = useCallback((messageId) => {
    const container = messagesContainerRef.current;

    const element = document.getElementById(
      `message-${messageId}`
    );

    if (!container || !element || !container.contains(element)) {
      return;
    }

    shouldScrollRef.current = false;

    const reducedMotion = window.matchMedia(
      "(prefers-reduced-motion: reduce)"
    ).matches;

    element.scrollIntoView({
      behavior: reducedMotion ? "instant" : "smooth",
      block: "center",
    });

    if (!reducedMotion) {
      element.animate?.(
        [
          {
            backgroundColor: "rgba(59, 130, 246, 0.3)",
          },
          {
            backgroundColor: "transparent",
          },
        ],
        {
          duration: 1500,
          easing: "ease-out",
        }
      );
    }
  }, []);

  /* -------------------- RENDER -------------------- */

  return (
    <div className="grid h-full min-h-0 min-w-0 grid-rows-[auto_minmax(0,1fr)_auto] overflow-hidden bg-gray-950">
      <ConversationHeader
        conversationId={conversationId}
        userId={userId}
        chat={chat}
        chatName={chatName}
        isGroup={isGroup}
        isPrivate={isPrivate}
        isBlockedByMe={isBlockedByMe}
        messagingUnavailable={messagingUnavailable}
        messages={messages}
        onJumpToMessage={jumpToMessage}
      />

      <MessageList
        messages={messages}
        loading={loading}
        userId={userId}
        isGroup={isGroup}
        canSendMessages={canSendMessages}
        recipientReadTime={recipientReadTime}
        replyMessages={replyMessages}
        reactionsByMessage={reactionsByMessage}
        getSenderName={getSenderName}
        onJumpToMessage={jumpToMessage}
        onToggleReaction={toggleReaction}
        containerRef={messagesContainerRef}
        onScroll={handleScroll}
      />

      <div className="min-w-0 border-t border-gray-800 bg-gray-950">
        {error && (
          <p role="alert" className="px-4 py-2 text-sm text-red-400">
            {error}
          </p>
        )}

        {reactionError && (
          <p role="alert" className="px-4 py-2 text-sm text-red-400">
            {reactionError}
          </p>
        )}

        {membershipError && (
          <p role="alert" className="px-4 py-2 text-xs text-amber-400">
            Unable to synchronise conversation members.
          </p>
        )}

        {canSendMessages ? (
          <>
            <TypingIndicator names={typingNames} />

            <MessageInput
              userId={userId}
              onTyping={notifyTyping}
              onStopTyping={stopTyping}
              replySenderName={getSenderName(replyTo)}
              mentionMembers={
                isGroup ? otherMembers : EMPTY_ARRAY
              }
            />

          </>
        ) : (
          <BlockedComposer
            isBlockedByMe={isBlockedByMe}
            messagingUnavailable={messagingUnavailable}
          />
        )}
      </div>
    </div>
  );
}

export default function ChatWindow(props) {
  return (
    <ChatSession
      key={`${props.userId}:${props.conversationId}`}
      {...props}
    />
  );
}

/* -------------------- MESSAGE TEXT -------------------- */

const MessageText = memo(function MessageText({
  parts,
  isOwn,
  hasMedia,
  userId,
}) {
  return (
    <p
      className={`whitespace-pre-wrap wrap-break-word [overflow-wrap:anywhere] text-sm ${hasMedia ? "mt-2" : ""
        }`}
    >
      {parts.map((part, index) => {
        if (part.type === "link") {
          return (
            <a
              key={index}
              href={part.url}
              target="_blank"
              rel="noopener noreferrer"
              className={`underline underline-offset-2 transition-colors ${isOwn
                ? "text-white hover:text-blue-200"
                : "text-blue-400 hover:text-blue-300"
                }`}
            >
              {part.value}
            </a>
          );
        }

        if (part.type === "mention") {
          const mentionsMe =
            part.userId === userId;

          return (
            <span
              key={index}
              title={
                mentionsMe
                  ? "Mentioned you"
                  : `Mentioned @${part.username}`
              }
              className={`rounded px-0.5 font-semibold ${mentionsMe
                ? "bg-amber-400/20 text-amber-300"
                : isOwn
                  ? "bg-white/15 text-white"
                  : "bg-blue-500/15 text-blue-300"
                }`}
            >
              {part.value}
            </span>
          );
        }

        return part.value;
      })}
    </p>
  );
});

/* -------------------- TIMESTAMP -------------------- */

  createdAt,
  isOwn,
  isRead,
  showReadStatus,
}) {

    const date = new Date(createdAt);

    return Number.isNaN(date.getTime())
      ? ""
      : TIME_FORMATTER.format(date);
  }, [createdAt]);

  return (
    <div
      className={`mt-1 flex items-center justify-end gap-1 text-[11px] ${isOwn ? "text-blue-200" : "text-gray-400"
        }`}
    >
      <time dateTime={createdAt}>{time}</time>

      {isOwn && showReadStatus && (
        isRead ? (
          <CheckCheck
            size={15}
            className="text-cyan-300"
            aria-label="Read"
          />
        ) : (
          <Check size={15} aria-label="Sent" />
        )
      )}
    </div>
  );
});

  quotedSenderName,
  onReply,
  onJumpToMessage,
  reactions = EMPTY_REACTIONS,

  onToggleReaction,
}) {
  const {

    content,
    mentions,
    created_at: createdAt,

const EMPTY_MEMBERS = Object.freeze([]);
const EMPTY_PROFILES = Object.freeze({});

/* -------------------- HELPERS -------------------- */

function getTimestamp(value) {
  const timestamp = Date.parse(value);

  return Number.isFinite(timestamp)
    ? timestamp
    : 0;
}

function latestReadAt(previous, incoming) {
  return getTimestamp(incoming) > getTimestamp(previous)
    ? incoming
    : previous ?? null;
}

function sameProfile(a, b) {
  return (
    a === b ||
    (
      a &&
      b &&
      a.id === b.id &&
      a.username === b.username &&
      a.display_name === b.display_name &&
      a.avatar_url === b.avatar_url
    )
  );
}

function reconcileMembers(
  previous,
  incoming,
  pendingReceipts
) {
  const previousById = new Map(
    previous.map((member) => [
      member.user_id,
      member,
    ])
  );

  const next = incoming.map((member) => {
    const old = previousById.get(member.user_id);

    const sameMembership =
      old?.joined_at === member.joined_at;

    const lastReadAt = latestReadAt(
      sameMembership ? old.last_read_at : null,
      latestReadAt(
        member.last_read_at,
        pendingReceipts.get(member.user_id)
      )
    );

    const profile = sameProfile(
      old?.profile,
      member.profile
    )
      ? old.profile
      : member.profile;

    if (
      old &&
      sameMembership &&
      old.last_read_at === lastReadAt &&
      old.profile === profile
    ) {
      return old;
    }

    return {
      ...member,
      last_read_at: lastReadAt,
      profile,
    };
  });

  return (
    previous.length === next.length &&
    next.every((member, index) =>
      member === previous[index]
    )
  )
    ? previous
    : next;
}

/* -------------------- HOOK -------------------- */

export default function useConversationMembers(
  conversationId,
  userId,
  enabled = true
) {
  const [snapshot, setSnapshot] = useState(null);

  useEffect(() => {
    if (!conversationId || !userId || !enabled) {
      return;
    }

    let active = true;
    let connected = false;
    let request = null;
    let refreshPending = false;

    const pendingReceipts = new Map();

    /* -------------------- REFRESH -------------------- */

    function refresh({ force = false } = {}) {
      if (!active) return;

      if (request) {
        if (force) {
          refreshPending = true;
        }

        return request.promise;
      }

      const controller = new AbortController();

      const currentRequest = {
        controller,
        promise: null,
      };

      request = currentRequest;

      currentRequest.promise = Promise.resolve()
        .then(async () => {
          do {
            refreshPending = false;

            if (!active) return;

            try {
              const incoming =
                await getConversationMembers(
                  conversationId,
                  controller.signal
                );

              if (
                !active ||
                controller.signal.aborted
              ) {
                return;
              }

              const receipts = new Map(
                pendingReceipts
              );

              pendingReceipts.clear();

              setSnapshot((current) => {
                const sameSession =
                  current?.conversationId ===
                    conversationId &&
                  current?.userId === userId;

                const previous = sameSession
                  ? current.members
                  : EMPTY_MEMBERS;

                const members = reconcileMembers(
                  previous,
                  incoming,
                  receipts
                );

                if (
                  sameSession &&
                  current.members === members &&
                  !current.error
                ) {
                  return current;
                }

                return {
                  conversationId,
                  userId,
                  members,
                  error: null,
                };
              });
            } catch (error) {
              if (
                !active ||
                controller.signal.aborted
              ) {
                return;
              }

              setSnapshot((current) => ({
                conversationId,
                userId,
                members:
                  current?.conversationId ===
                    conversationId &&
                  current?.userId === userId
                    ? current.members
                    : EMPTY_MEMBERS,
                error:
                  error?.message ||
                  "Unable to load conversation members",
              }));
            }
          } while (active && refreshPending);
        })
        .finally(() => {
          if (request === currentRequest) {
            request = null;
          }
        });

      return currentRequest.promise;
    }

    /* -------------------- REALTIME -------------------- */

    function handleReceipt(member) {
      if (
        !active ||
        member?.conversation_id !== conversationId ||
        !member.user_id ||
        !getTimestamp(member.last_read_at)
      ) {
        return;
      }

      if (request) {
        pendingReceipts.set(
          member.user_id,
          latestReadAt(
            pendingReceipts.get(member.user_id),
            member.last_read_at
          )
        );
      }

      setSnapshot((current) => {
        if (
          current?.conversationId !== conversationId ||
          current?.userId !== userId
        ) {
          return current;
        }

        const index = current.members.findIndex(
          (item) => item.user_id === member.user_id
        );

        if (index === -1) {
          return current;
        }

        const previous = current.members[index];

        const lastReadAt = latestReadAt(
          previous.last_read_at,
          member.last_read_at
        );

        if (
          lastReadAt === previous.last_read_at
        ) {
          return current;
        }

        const members = [...current.members];

        members[index] = {
          ...previous,
          last_read_at: lastReadAt,
        };

        return {
          ...current,
          members,
        };
      });
    }

    function handleStatus(status) {
      if (!active) return;

      if (status === "SUBSCRIBED") {
        connected = true;

        void refresh({ force: true });

        return;
      }

      if (
        status === "CHANNEL_ERROR" ||
        status === "TIMED_OUT" ||
        status === "CLOSED"
      ) {
        connected = false;
      }
    }

    const unsubscribe = subscribeToReadReceipts(
      conversationId,
      handleReceipt,
      handleStatus
    );

    /* -------------------- BROWSER EVENTS -------------------- */

    const unsubscribeVisibility =
      subscribePageVisibility(() => {
        if (active && getPageVisible()) {
          void refresh();
        }
      });

    function handleOnline() {
      if (
        active &&
        !connected &&
        getPageVisible()
      ) {
        void refresh();
      }
    }

    window.addEventListener(
      "online",
      handleOnline
    );

    void refresh();

    /* -------------------- CLEANUP -------------------- */

    return () => {
      active = false;
      refreshPending = false;

      request?.controller.abort();

      pendingReceipts.clear();

      unsubscribe();
      unsubscribeVisibility();

      window.removeEventListener(
        "online",
        handleOnline
      );
    };
  }, [conversationId, userId, enabled]);

  /* -------------------- DERIVED STATE -------------------- */

  const current =
    enabled &&
    snapshot?.conversationId === conversationId &&
    snapshot?.userId === userId
      ? snapshot
      : null;

  const members = current?.members ?? EMPTY_MEMBERS;

  const otherMembers = useMemo(
    () => members.filter(
      (member) => member.user_id !== userId
    ),
    [members, userId]
  );

  const profiles = useMemo(() => {
    if (!members.length) {
      return EMPTY_PROFILES;
    }

    return Object.fromEntries(
      members
        .filter((member) => member.profile)
        .map((member) => [
          member.user_id,
          member.profile,
        ])
    );
  }, [members]);

  const recipientReadTime =
    otherMembers.length === 1
      ? getTimestamp(otherMembers[0].last_read_at)
      : 0;

  return {
    members,
    otherMembers,
    profiles,
    recipientReadTime,
    loading: Boolean(
      enabled &&
      conversationId &&
      userId &&
      !current
    ),
    error: current?.error ?? null,
  };
}

    throw new Error("Invalid GIF ID");

import { createClient } from "@/lib/supabase/client";

const supabase = createClient();

const CACHE_TTL = 5 * 60 * 1000;
const MISSING_TTL = 30 * 1000;

const MAX_CACHE_SIZE = 1000;
const BATCH_SIZE = 100;

const PROFILE_COLUMNS = "id, username, display_name";

const profileCache = new Map();
const pendingRequests = new Map();

let queuedProfiles = new Map();
let batchScheduled = false;
let generation = 0;

/* -------------------- CACHE -------------------- */

function getCachedProfile(id) {
  const entry = profileCache.get(id);

  if (!entry) return undefined;

  if (entry.expiresAt <= Date.now()) {
    profileCache.delete(id);
    return undefined;
  }

  // Move recently accessed profiles to the end.
  profileCache.delete(id);
  profileCache.set(id, entry);

  return entry.profile;
}

function cacheProfile(id, profile, now) {
  profileCache.delete(id);

  profileCache.set(id, {
    profile,
    expiresAt:
      now + (profile ? CACHE_TTL : MISSING_TTL),
  });

  while (profileCache.size > MAX_CACHE_SIZE) {
    const oldest = profileCache.keys().next().value;

    profileCache.delete(oldest);
  }
}

/* -------------------- PROFILE RETRIEVAL -------------------- */

async function fetchBatch(entries, currentGeneration) {
  const ids = entries.map(([id]) => id);

  try {
    if (currentGeneration !== generation) {
      for (const [, request] of entries) {
        request.resolve(null);
      }

      return;
    }

    const { data, error } = await supabase
      .from("profiles")
      .select(PROFILE_COLUMNS)
      .in("id", ids);

    if (error) throw error;

    const profiles = new Map(
      (data ?? []).map((profile) => [
        profile.id,
        profile,
      ])
    );

    const isCurrent =
      currentGeneration === generation;

    const now = Date.now();

    for (const [id, request] of entries) {
      const profile = isCurrent
        ? profiles.get(id) ?? null
        : null;

      if (isCurrent) {
        cacheProfile(id, profile, now);
      }

      request.resolve(profile);
    }
  } catch (error) {
    for (const [, request] of entries) {
      if (currentGeneration === generation) {
        request.reject(error);
      } else {
        request.resolve(null);
      }
    }
  } finally {
    for (const [id, request] of entries) {
      if (pendingRequests.get(id) === request.promise) {
        pendingRequests.delete(id);
      }
    }
  }
}

/* -------------------- REQUEST BATCHING -------------------- */

async function flushQueue(currentGeneration) {
  if (currentGeneration !== generation) return;

  batchScheduled = false;

  const entries = [...queuedProfiles];

  queuedProfiles.clear();

  // Process bounded batches sequentially.
  for (
    let offset = 0;
    offset < entries.length;
    offset += BATCH_SIZE
  ) {
    await fetchBatch(
      entries.slice(offset, offset + BATCH_SIZE),
      currentGeneration
    );
  }
}

function scheduleBatch() {
  if (batchScheduled) return;

  batchScheduled = true;

  const currentGeneration = generation;

  queueMicrotask(() => {
    void flushQueue(currentGeneration);
  });
}

function requestProfile(id) {
  const cached = getCachedProfile(id);

  // null represents a cached missing profile.
  if (cached !== undefined) {
    return Promise.resolve(cached);
  }

  const pending = pendingRequests.get(id);

  if (pending) return pending;

  let resolve;
  let reject;

  const promise = new Promise((res, rej) => {
    resolve = res;
    reject = rej;
  });

  pendingRequests.set(id, promise);

  queuedProfiles.set(id, {
    promise,
    resolve,
    reject,
  });

  scheduleBatch();

  return promise;
}

/* -------------------- PUBLIC INTERFACE -------------------- */

export async function getMessageProfiles(userIds) {
  const currentGeneration = generation;

  const ids = [
    ...new Set(userIds.filter(Boolean)),
  ];

  if (!ids.length) return {};

  const profiles = await Promise.all(
    ids.map(requestProfile)
  );

  if (currentGeneration !== generation) {
    return {};
  }

  return Object.fromEntries(
    ids.map((id, index) => [
      id,
      profiles[index],
    ])
  );
}

export function clearMessageProfileCache() {
  generation++;

  // Settle requests that have not started yet.
  for (const request of queuedProfiles.values()) {
    request.resolve(null);
  }

  queuedProfiles = new Map();
  batchScheduled = false;

  profileCache.clear();
  pendingRequests.clear();
}

/* -------------------- DATES -------------------- */

export function getDateKey(value) {
  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return "unknown";
  }

  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");

  return `${year}-${month}-${day}`;
}

export function formatMessageDate(value, now = new Date()) {
  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return "Unknown date";
  }

  const key = getDateKey(date);

  if (key === getDateKey(now)) {
    return "Today";
  }

  const yesterday = new Date(now);

  yesterday.setDate(yesterday.getDate() - 1);

  return key === getDateKey(yesterday)
    ? "Yesterday"
    : dateFormatter.format(date);
}

export function groupMessagesByDate(messages) {
  const groups = [];

  for (const message of messages) {
    const key = getDateKey(message.created_at);
    const previous = groups.at(-1);

    if (previous?.key === key) {
      previous.messages.push(message);
    } else {
      groups.push({
        key,
        date: message.created_at,
        messages: [message],
      });
    }
  }

  return groups;
}

/* -------------------- TEXT -------------------- */

export function truncateText(
  value,
  maxLength = MAX_NOTIFICATION_LENGTH
) {
  if (
    typeof value !== "string" ||
    !Number.isFinite(maxLength) ||
    maxLength < 1
  ) {
    return "";
  }

  const limit = Math.floor(maxLength);

  if (value.length <= limit) {
    return value;
  }

  let count = 0;
  let offset = 0;
  let truncateAt = 0;

  for (const character of value) {
    if (count === limit) {
      return value.slice(0, truncateAt) + "…";
    }

    offset += character.length;
    count++;

    if (count < limit) {
      truncateAt = offset;
    }
  }

  return value;
}

/* -------------------- COMPOSITION -------------------- */

export function insertEmoji(
  text,
  emoji,
  selectionStart,
  selectionEnd,
  maxLength = MAX_MESSAGE_LENGTH
) {
  const value = typeof text === "string" ? text : "";

  const clamp = (position) =>
    Number.isInteger(position)
      ? Math.max(0, Math.min(position, value.length))
      : value.length;

  const start = clamp(selectionStart);
  const end = Math.max(start, clamp(selectionEnd));

  const unchanged = {
    text: value,
    cursor: start,
    inserted: false,
  };

  if (typeof emoji !== "string" || !emoji) {
    return unchanged;
  }

  const nextLength =
    value.length - (end - start) + emoji.length;

  if (nextLength > maxLength) {
    return unchanged;
  }

  return {
    text:
      value.slice(0, start) +
      emoji +
      value.slice(end),

    cursor: start + emoji.length,
    inserted: true,
  };
}

/* -------------------- MESSAGE PREVIEWS -------------------- */

function formatPreview(message, notification = false) {
  const fallback = notification
    ? "Sent you a message"
    : "Message";

  if (!message) {
    return notification
      ? fallback
      : "Message unavailable";
  }

  const content = message.content?.trim() || "";

  if (message.gif_id) {
    return content
      ? `GIF: ${content}`
      : notification
        ? "Sent you a GIF"
        : "GIF";
  }

  if (message.attachment_type === "image") {
    return content
      ? `Photo: ${content}`
      : notification
        ? "Sent you a photo"
        : "Photo";
  }

  if (message.attachment_type === "file") {
    if (!notification) {
      return (
        content ||
        message.attachment_name ||
        "Attachment"
      );
    }

    if (content) {
      return `File: ${content}`;
    }

    const name = message.attachment_name?.trim();

    return name
      ? `Sent you a file: ${name}`
      : "Sent you a file";
  }

  return content || fallback;
}

export function getMessagePreview(message) {
  return formatPreview(message);
}

export function getNotificationPreview(message) {
  return truncateText(
    formatPreview(message, true),
    MAX_NOTIFICATION_LENGTH
  );
}

/* -------------------- MENTIONS -------------------- */

const MENTION_QUERY_PATTERN =
  /(?:^|[\s([{])@([A-Za-z0-9_]*)$/;

const USERNAME_PATTERN = /^[A-Za-z0-9_]+$/;

export function getMentionQuery(
  text,
  selectionStart,
  selectionEnd = selectionStart
) {
  if (
    typeof text !== "string" ||
    !Number.isInteger(selectionStart) ||
    selectionStart !== selectionEnd ||
    selectionStart < 0 ||
    selectionStart > text.length
  ) {
    return null;
  }

  const beforeCursor = text.slice(
    0,
    selectionStart
  );

  const match = MENTION_QUERY_PATTERN.exec(
    beforeCursor
  );

  if (!match) return null;

  const query = match[1];

  // Include the remainder of an existing username
  // when editing a mention in the middle of a message.
  const remaining = text
    .slice(selectionStart)
    .match(/^[A-Za-z0-9_]*/)?.[0] ?? "";

  return {
    query,
    start: selectionStart - query.length - 1,
    end: selectionStart + remaining.length,
  };
}

export function getMentionSuggestions(
  members,
  query = ""
) {
  const search = query.toLowerCase();

  const matches = [];

  for (const member of members) {
    const profile = member.profile;
    const username = profile?.username;

    if (
      typeof username !== "string" ||
      !USERNAME_PATTERN.test(username)
    ) {
      continue;
    }

    const normalizedUsername =
      username.toLowerCase();

    const normalizedName =
      profile.display_name?.toLowerCase() ?? "";

    let score;

    if (normalizedUsername === search) {
      score = 0;
    } else if (
      normalizedUsername.startsWith(search)
    ) {
      score = 1;
    } else if (
      normalizedName.startsWith(search)
    ) {
      score = 2;
    } else if (
      normalizedUsername.includes(search) ||
      normalizedName.includes(search)
    ) {
      score = 3;
    } else {
      continue;
    }

    matches.push({
      profile,
      score,
    });
  }

  return matches
    .sort(
      (a, b) =>
        a.score - b.score ||
        a.profile.username.localeCompare(
          b.profile.username
        )
    )
    .slice(0, MAX_MENTION_SUGGESTIONS)
    .map(({ profile }) => profile);
}

export function insertMention(
  text,
  username,
  mention,
  maxLength = MAX_MESSAGE_LENGTH
) {
  if (
    typeof text !== "string" ||
    typeof username !== "string" ||
    !USERNAME_PATTERN.test(username) ||
    !mention ||
    !Number.isInteger(mention.start) ||
    !Number.isInteger(mention.end) ||
    mention.start < 0 ||
    mention.end < mention.start ||
    mention.end > text.length ||
    text[mention.start] !== "@"
  ) {
    return null;
  }

  const before = text.slice(
    0,
    mention.start
  );

  const after = text.slice(
    mention.end
  );

  const token = `@${username}`;

  const separator =
    /^\s/.test(after) ? "" : " ";

  const inserted = token + separator;

  const nextText =
    before + inserted + after;

  if (nextText.length > maxLength) {
    return null;
  }

  return {
    text: nextText,
    cursor:
      before.length +
      inserted.length +
      (separator ? 0 : 1),
  };
}

/* -------------------- MENTION HIGHLIGHTING -------------------- */

const MENTION_TOKEN_PATTERN =
  /(^|[\s([{])@([A-Za-z0-9_]+)(?![A-Za-z0-9_])/g;

export function highlightMentionParts(
  parts,
  mentions
) {
  if (
    !Array.isArray(mentions) ||
    !mentions.length ||
    !parts.length
  ) {
    return parts;
  }

  const mentionsByUsername = new Map();

  for (const mention of mentions) {
    if (
      typeof mention?.username !== "string" ||
      typeof mention.user_id !== "string" ||
      !USERNAME_PATTERN.test(mention.username)
    ) {
      continue;
    }

    mentionsByUsername.set(
      mention.username,
      mention
    );
  }

  if (!mentionsByUsername.size) {
    return parts;
  }

  const result = [];

  let changed = false;

  for (const part of parts) {
    // Preserve links and other existing part types.
    if (
      part.type !== "text" ||
      !part.value.includes("@")
    ) {
      result.push(part);
      continue;
    }

    let offset = 0;

    for (const match of part.value.matchAll(
      MENTION_TOKEN_PATTERN
    )) {
      const prefix = match[1];
      const username = match[2];

      const mention =
        mentionsByUsername.get(username);

      if (!mention) continue;

      const start =
        match.index + prefix.length;

      const end =
        start + username.length + 1;

      if (start > offset) {
        result.push({
          type: "text",
          value: part.value.slice(
            offset,
            start
          ),
        });
      }

      result.push({
        type: "mention",
        value: part.value.slice(start, end),
        userId: mention.user_id,
        username,
      });

      offset = end;
      changed = true;
    }

    if (offset < part.value.length) {
      result.push({
        type: "text",
        value: part.value.slice(offset),
      });
    }
  }

  return changed ? result : parts;
}
