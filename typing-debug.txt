"use client";

import {
  useCallback,
  useEffect,
  useRef,
  useState,
} from "react";

import { createTypingChannel } from "@/lib/extras/typing";

const BROADCAST_INTERVAL = 3000;
const INACTIVITY_TIMEOUT = 2000;
const REMOTE_TIMEOUT = 7000;

const EMPTY_USERS = Object.freeze([]);

export default function useTypingIndicator(
  conversationId,
  userId,
  { enabled = true } = {}
) {
  const sessionRef = useRef(null);
  const [typingState, setTypingState] = useState(null);
  const canUseTyping = Boolean(enabled && conversationId && userId);

  /* -------------------- PUBLIC ACTIONS -------------------- */

  const notifyTyping = useCallback(
    (value) => {
      const session = sessionRef.current;

      if (
        !canUseTyping ||
        session?.conversationId !== conversationId ||
        session?.userId !== userId
      ) {
        return;
      }

      session.notify(value);
    },
    [canUseTyping, conversationId, userId]
  );

  const stopTyping = useCallback(() => {
    const session = sessionRef.current;

    if (
      session?.conversationId === conversationId &&
      session?.userId === userId
    ) {
      session.stop();
    }
  }, [conversationId, userId]);

  /* -------------------- SESSION -------------------- */

  useEffect(() => {
    if (!canUseTyping) return;

    let active = true;
    let connection = null;

    let localTyping = false;
    let lastBroadcastAt = null;
    let inactivityTimer = null;

    const remoteTimers = new Map();

    const session = {
      conversationId,
      userId,
      notify,
      stop,
    };

    sessionRef.current = session;

    /* -------------------- LOCAL TYPING -------------------- */

    function broadcastTyping() {
      if (!active || !localTyping) return;

      const now = Date.now();

      if (
        lastBroadcastAt !== null &&
        now - lastBroadcastAt < BROADCAST_INTERVAL
      ) {
        return;
      }

      if (connection?.broadcast(true)) {
        lastBroadcastAt = now;
      }
    }

    function stop() {
      clearTimeout(inactivityTimer);
      inactivityTimer = null;

      if (!localTyping) return;

      localTyping = false;
      lastBroadcastAt = null;

      connection?.broadcast(false);
    }

    function notify(value) {
      if (!active) return;

      if (
        typeof value !== "string" ||
        !value.trim() ||
        document.visibilityState === "hidden"
      ) {
        stop();
        return;
      }

      localTyping = true;

      broadcastTyping();

      clearTimeout(inactivityTimer);

      inactivityTimer = setTimeout(
        stop,
        INACTIVITY_TIMEOUT
      );
    }

    /* -------------------- REMOTE TYPING -------------------- */

    function publishUsers() {
      if (!active) return;

      setTypingState({
        session,
        conversationId,
        userId,
        users: remoteTimers.size
          ? [...remoteTimers.keys()]
          : EMPTY_USERS,
      });
    }

    function removeUser(id) {
      if (!remoteTimers.has(id)) return;

      clearTimeout(remoteTimers.get(id));

      remoteTimers.delete(id);

      publishUsers();
    }

    function clearRemoteUsers(publish = true) {
      if (!remoteTimers.size) return;

      for (const timer of remoteTimers.values()) {
        clearTimeout(timer);
      }

      remoteTimers.clear();

      if (publish) {
        publishUsers();
      }
    }

    function handleTyping(payload) {
      if (
        !active ||
        !payload ||
        typeof payload.userId !== "string" ||
        payload.userId === userId ||
        typeof payload.typing !== "boolean"
      ) {
        return;
      }

      const {
        userId: senderId,
        typing,
      } = payload;

      if (!typing) {
        removeUser(senderId);
        return;
      }

      const alreadyTyping =
        remoteTimers.has(senderId);

      clearTimeout(remoteTimers.get(senderId));

      remoteTimers.set(
        senderId,
        setTimeout(() => {
          removeUser(senderId);
        }, REMOTE_TIMEOUT)
      );

      // Heartbeats only extend the expiry timer.
      // They do not need to update React state.
      if (!alreadyTyping) {
        publishUsers();
      }
    }

    /* -------------------- CONNECTION -------------------- */

    function handleStatus(connected) {
      if (!active) return;

      lastBroadcastAt = null;

      if (connected) {
        broadcastTyping();
      } else {
        clearRemoteUsers();
      }
    }

    async function connect() {
      try {
        const result = await createTypingChannel({
          conversationId,
          userId,
          onTyping: handleTyping,
          onStatus: handleStatus,
        });

        if (!active) {
          result.close();
          return;
        }

        connection = result;

        // Handles a subscription becoming ready
        // before the asynchronous setup completes.
        if (connection.isConnected()) {
          broadcastTyping();
        }
      } catch (error) {
        if (active) {
          console.error(
            "Failed to connect typing channel:",
            error
          );
        }
      }
    }

    /* -------------------- VISIBILITY -------------------- */

    function handleVisibility() {
      if (document.visibilityState === "hidden") {
        stop();
      }
    }

    document.addEventListener(
      "visibilitychange",
      handleVisibility
    );

    void connect();

    /* -------------------- CLEANUP -------------------- */

    return () => {
      active = false;

      stop();

      clearRemoteUsers(false);

      connection?.close();

      if (sessionRef.current === session) {
        sessionRef.current = null;
      }

      document.removeEventListener(
        "visibilitychange",
        handleVisibility
      );
    };
  }, [conversationId, userId, canUseTyping]);

  /* -------------------- VISIBLE USERS -------------------- */

  const typingUsers =
    canUseTyping &&
    typingState?.conversationId === conversationId &&
    typingState?.userId === userId
      ? typingState.users
      : EMPTY_USERS;

  return {
    typingUsers,
    notifyTyping,
    stopTyping,
  };
}

import { createClient } from "@/lib/supabase/client";
import { isUuid } from "@/utils/validation";

const supabase = createClient();

export async function createTypingChannel({
  conversationId,
  userId,
  onTyping,
  onStatus,
}) {
  if (!isUuid(conversationId) || !isUuid(userId)) {
    throw new Error("Invalid conversation or user ID");
  }

  await supabase.realtime.setAuth();

  let closed = false;
  let connected = false;
  let advertisedTyping = false;

  const channel = supabase.channel(`typing:${conversationId}`,
    {
      config: {
        private: true,
        broadcast: {
          self: false,
          ack: false,
        },
      },
    }
  );

  function handleTyping({ payload }) {
    if (
      closed ||
      !connected ||
      !isUuid(payload?.userId) ||
      payload.userId === userId ||
      typeof payload.typing !== "boolean"
    ) {
      return;
    }

    onTyping?.({
      userId: payload.userId,
      typing: payload.typing,
    });
  }

  function handleStatus(status, error) {
    if (closed) return;

    const nextConnected = status === "SUBSCRIBED";

    if (connected !== nextConnected) {
      connected = nextConnected;

      if (!connected) {
        advertisedTyping = false;
      }

      onStatus?.(connected);
    }

    if (
      status === "CHANNEL_ERROR" ||
      status === "TIMED_OUT"
    ) {
      console.warn("Typing channel:", status, error);
    }
  }

  function broadcast(typing) {
    if (
      closed ||
      !connected ||
      typeof typing !== "boolean"
    ) {
      return false;
    }

    // Repeated true events are intentional heartbeats.
    // Repeated false events are unnecessary.
    if (!typing && !advertisedTyping) {
      return false;
    }

    advertisedTyping = typing;

    void channel
      .send({
        type: "broadcast",
        event: "typing",
        payload: {
          userId,
          typing,
        },
      })
      .then((status) => {
        if (!closed && status !== "ok") {
          console.warn(
            "Typing broadcast status:",
            status
          );
        }
      })
      .catch((error) => {
        if (!closed) {
          console.warn(
            "Typing broadcast failed:",
            error
          );
        }
      });

    return true;
  }

  function close() {
    if (closed) return;

    // Only sends false if typing was advertised.
    broadcast(false);

    closed = true;
    connected = false;

    void supabase.removeChannel(channel);
  }

  try {
    channel
      .on(
        "broadcast",
        { event: "typing" },
        handleTyping
      )
      .subscribe(handleStatus);
  } catch (error) {
    close();
    throw error;
  }

  return {
    broadcast,
    close,

    isConnected() {
      return connected && !closed;
    },
  };
}

    "setTimeout"
  );

  const clearTimeoutSpy = jest.spyOn(
    globalThis,
    "clearTimeout"
  );

  try {
    act(() => {
      result.current.notifyTyping("Hello");
    });

    receiveTyping(USER_B, true);
    receiveTyping(USER_C, true);

    // Capture only timers associated with typing.
    const typingTimers = setTimeoutSpy.mock.calls
      .map((args, index) => ({
        delay: args[1],
        handle: setTimeoutSpy.mock.results[index].value,
      }))
      .filter(
        ({ delay }) =>
          delay === INACTIVITY_TIMEOUT ||
          delay === REMOTE_TIMEOUT
      );

    // One local timer and two remote timers.
    expect(typingTimers).toHaveLength(3);

    expect(
      typingTimers.filter(
        ({ delay }) => delay === INACTIVITY_TIMEOUT
      )
    ).toHaveLength(1);

    expect(typingTimers.filter(({ delay }) => delay === REMOTE_TIMEOUT)).toHaveLength(2);

    clearTimeoutSpy.mockClear();

    unmount();

    // Verify every typing timer was explicitly cleared.
    for (const { handle } of typingTimers) {
      expect(clearTimeoutSpy).toHaveBeenCalledWith(handle);
    }

    expect(connection.close).toHaveBeenCalledTimes(1);

    connection.broadcast.mockClear();

    await advanceTime(REMOTE_TIMEOUT * 2);

    // No delayed broadcasts after unmounting.
    expect(connection.broadcast).not.toHaveBeenCalled();

  } finally {
    setTimeoutSpy.mockRestore();
    clearTimeoutSpy.mockRestore();
  }
});
