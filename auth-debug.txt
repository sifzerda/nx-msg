"use client";

import { useRef, useState } from "react";
import Link from "next/link";
import { usePathname } from "next/navigation";

import {
  MessageCircle,
  Users,
  LogOut,
  LogIn,
  UserPlus,
  Settings,
  LoaderCircle,
} from "lucide-react";

import { logout } from "@/lib/logout";

const AUTHENTICATED_LINKS = [
  {
    href: "/messages",
    label: "Messages",
    icon: MessageCircle,
  },
  {
    href: "/contacts",
    label: "Contacts",
    icon: Users,
  },
  {
    href: "/settings",
    label: "Settings",
    icon: Settings,
  },
];

const GUEST_LINKS = [
  {
    href: "/login",
    label: "Login",
    icon: LogIn,
  },
  {
    href: "/register",
    label: "Register",
    icon: UserPlus,
  },
];

const NAV_LINK_CLASS =
  "flex shrink-0 items-center gap-2 rounded-lg px-3 py-2 text-sm transition-colors";

export default function Header({ user = null }) {
  const pathname = usePathname() ?? "";

  const logoutRef = useRef(false);

  const [loggingOut, setLoggingOut] =
    useState(false);

  const [error, setError] = useState("");

  const links = user
    ? AUTHENTICATED_LINKS
    : GUEST_LINKS;

  async function handleLogout() {
    if (logoutRef.current) return;

    logoutRef.current = true;

    setLoggingOut(true);
    setError("");

    let navigating = false;

    try {
      const result = await logout();

      if (result.pushCleanupError) {
        console.warn(
          "Push cleanup failed:",
          result.pushCleanupError
        );
      }

      window.location.replace("/login");

      navigating = true;
    } catch (error) {
      console.error("Logout failed:", error);

      setError(
        error?.message || "Could not log out."
      );
    } finally {
      if (!navigating) {
        logoutRef.current = false;
        setLoggingOut(false);
      }
    }
  }

  return (
    <header className="relative flex h-16 shrink-0 items-center justify-between gap-2 border-b border-gray-800 bg-gray-900 px-3 sm:gap-4 sm:px-4 md:px-6">
      <Link
        href={user ? "/messages" : "/login"}
        prefetch={false}
        aria-label="Messenger home"
        className="flex shrink-0 items-center gap-2 text-lg font-bold text-white"
      >
        <MessageCircle
          size={24}
          className="text-blue-400"
          aria-hidden="true"
        />

        <span className="hidden lg:inline">
          Messenger
        </span>
      </Link>

      <nav
        aria-label="Main navigation"
        className="flex min-w-0 items-center gap-1"
      >
        {links.map(({ href, label, icon: Icon }) => {
          const active =
            pathname === href ||
            pathname.startsWith(`${href}/`);

          return (
            <Link
              key={href}
              href={href}
              prefetch={false}
              title={label}
              aria-label={label}
              aria-current={
                active ? "page" : undefined
              }
              className={`${NAV_LINK_CLASS} ${
                active
                  ? "bg-blue-600 text-white"
                  : "text-gray-400 hover:bg-gray-800 hover:text-white"
              }`}
            >
              <Icon
                size={18}
                className="shrink-0"
                aria-hidden="true"
              />

              <span className="hidden md:inline">
                {label}
              </span>
            </Link>
          );
        })}
      </nav>

      {user && (
        <div className="flex min-w-0 items-center gap-3">
          <div className="hidden min-w-0 text-right lg:block">
            <p className="max-w-40 truncate text-sm font-medium text-white">
              {user.display_name || "User"}
            </p>

            <p className="max-w-40 truncate text-xs text-gray-400">
              @{user.username || "unknown"}
            </p>
          </div>

          <button
            type="button"
            onClick={handleLogout}
            disabled={loggingOut}
            title={
              loggingOut
                ? "Logging out..."
                : "Log out"
            }
            aria-label={
              loggingOut
                ? "Logging out"
                : "Log out"
            }
            aria-busy={loggingOut}
            className="flex shrink-0 items-center gap-2 rounded-lg border border-gray-700 px-3 py-2 text-sm text-gray-300 transition-colors hover:border-red-500 hover:bg-red-500/10 hover:text-red-400 disabled:cursor-not-allowed disabled:opacity-50"
          >
            {loggingOut ? (
              <LoaderCircle
                size={18}
                className="animate-spin"
                aria-hidden="true"
              />
            ) : (
              <LogOut
                size={18}
                aria-hidden="true"
              />
            )}

            <span className="hidden lg:inline">
              {loggingOut
                ? "Logging out..."
                : "Logout"}
            </span>
          </button>
        </div>
      )}

      {error && (
        <div
          role="alert"
          className="absolute right-3 top-full z-50 mt-2 max-w-sm rounded-lg border border-red-500/30 bg-gray-900 p-3 text-sm text-red-400 shadow-lg"
        >
          {error}
        </div>
      )}
    </header>
  );
}

"use client";

import { useEffect } from "react";

import { createClient } from "@/lib/supabase/client";

import {
  resetApplication,
  isLoggingOut,
  SESSION_CHANGE_KEY,
} from "@/lib/logout";

import usePushStore from "@/store/usePushStore";

export default function SessionInitializer({
  userId = null,
}) {
  useEffect(() => {
    const supabase = createClient();

    let active = true;
    let reloading = false;
    let reloadTimer = null;

    /* -------------------- SESSION CHANGE -------------------- */

    function reloadSession() {
      if (
        !active ||
        reloading ||
        isLoggingOut()
      ) {
        return;
      }

      reloading = true;

      // Run outside the Supabase authentication callback.
      reloadTimer = window.setTimeout(() => {
        if (!active) return;

        if (isLoggingOut()) {
          reloading = false;
          return;
        }

        try {
          resetApplication();
        } finally {
          window.location.reload();
        }
      }, 0);
    }

    /* -------------------- AUTHENTICATION EVENTS -------------------- */

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange(
      (event, session) => {
        if (!active || reloading) return;

        if (
          event === "INITIAL_SESSION" ||
          event === "TOKEN_REFRESHED"
        ) {
          return;
        }

        if (
          event !== "SIGNED_IN" &&
          event !== "SIGNED_OUT"
        ) {
          return;
        }

        const nextUserId =
          session?.user?.id ?? null;

        if (nextUserId !== userId) {
          reloadSession();
        }
      }
    );

    /* -------------------- CROSS-TAB EVENTS -------------------- */

    function handleStorage(event) {
      if (
        event.key === SESSION_CHANGE_KEY &&
        event.newValue
      ) {
        reloadSession();
      }
    }

    window.addEventListener(
      "storage",
      handleStorage
    );

    /* -------------------- PUSH INITIALISATION -------------------- */

    const pushStore = usePushStore.getState();

    if (userId) {
      void pushStore.initialize(userId);
    } else {
      pushStore.reset();
    }

    /* -------------------- CLEANUP -------------------- */

    return () => {
      active = false;

      subscription.unsubscribe();

      window.removeEventListener(
        "storage",
        handleStorage
      );

      if (reloadTimer !== null) {
        window.clearTimeout(reloadTimer);
      }
    };
  }, [userId]);

  return null;
}

"use client";

import { createClient } from "@/lib/supabase/client";
import { cleanupPushForLogout } from "@/lib/pushClient";

import { clearMessageProfileCache } from "@/lib/messageProfiles";
import { clearAttachmentCache } from "@/lib/extras/attachments";
import { clearLinkPreviewCache } from "@/lib/extras/linkPreviews";

import usePushStore from "@/store/usePushStore";
import useChatStore from "@/store/useChatStore";
import useContactStore from "@/store/useContactStore";
import usePresenceStore from "@/store/usePresenceStore";
import useBlockStore from "@/store/useBlockStore";

let logoutRequest = null;

/* -------------------- APPLICATION CLEANUP -------------------- */

export function resetApplication() {
    usePushStore.getState().reset();
    useBlockStore.getState().reset();
    useChatStore.getState().reset();
    useContactStore.getState().reset();
    usePresenceStore.getState().reset();

    clearMessageProfileCache();
    clearAttachmentCache();
    clearLinkPreviewCache();
}

export function isLoggingOut() {
    return logoutRequest !== null;
}

/* -------------------- SESSION SYNCHRONISATION -------------------- */

export const SESSION_CHANGE_KEY =
    "messenger:session-change";

export function notifySessionChange() {
    try {
        window.localStorage.setItem(
            SESSION_CHANGE_KEY,
            crypto.randomUUID()
        );
    } catch {
        // Storage may be unavailable or restricted.
    }
}

/* -------------------- PUSH CLEANUP -------------------- */

async function prepareNotifications() {
    let cleanupError = null;

    try {
        await usePushStore.getState().prepareForLogout();
    } catch (error) {
        cleanupError = error;
    }

    try {
        const error = await cleanupPushForLogout();
        cleanupError ??= error;
    } catch (error) {
        cleanupError ??= error;
    }

    if (cleanupError) {
        console.error(
            "Push logout cleanup failed:",
            cleanupError.message
        );
    }

    return cleanupError;
}

/* -------------------- RECOVERY -------------------- */

function restoreNotifications() {
    const store = usePushStore.getState();

    store.cancelLogout();

    void store.refresh().catch((error) => {
        console.error(
            "Failed to restore notifications:",
            error?.message
        );
    });
}

/* -------------------- LOGOUT -------------------- */

async function performLogout() {
    const supabase = createClient();

    let signedOut = false;

    try {
        const pushCleanupError =
            await prepareNotifications();

        const { error } =
            await supabase.auth.signOut();

        if (error) throw error;

        signedOut = true;

        resetApplication();
        notifySessionChange();

        return {
            success: true,
            pushCleanupError,
        };
    } catch (error) {
        if (!signedOut) {
            restoreNotifications();
        }

        throw error;
    }
}

/* -------------------- PUBLIC API -------------------- */

export function logout() {
    if (logoutRequest) {
        return logoutRequest;
    }

    const request = Promise.resolve()
        .then(performLogout)
        .finally(() => {
            if (logoutRequest === request) {
                logoutRequest = null;
            }
        });

    logoutRequest = request;

    return request;
}

// src/lib/pushClient.js

import { createClient } from "@/lib/supabase/client";

const SERVICE_WORKER_URL = "/sw.js";

const VAPID_PUBLIC_KEY =
  process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY;

const DISABLED_STATE = Object.freeze({
  preference: false,
  enabled: false,
});

const ENABLED_STATE = Object.freeze({
  preference: true,
  enabled: true,
});

let registrationRequest = null;

/* -------------------- SUPPORT -------------------- */

export function isPushSupported() {
  return (
    typeof window !== "undefined" &&
    typeof navigator !== "undefined" &&
    window.isSecureContext &&
    "serviceWorker" in navigator &&
    "PushManager" in window &&
    "Notification" in window
  );
}

/* -------------------- VAPID -------------------- */

function getApplicationServerKey() {
  if (!VAPID_PUBLIC_KEY) {
    throw new Error("Missing VAPID public key");
  }

  const padding = "=".repeat(
    (4 - (VAPID_PUBLIC_KEY.length % 4)) % 4
  );

  const base64 = (VAPID_PUBLIC_KEY + padding)
    .replace(/-/g, "+")
    .replace(/_/g, "/");

  return Uint8Array.from(
    atob(base64),
    (char) => char.charCodeAt(0)
  );
}

function matchesApplicationServerKey(
  subscription,
  expectedKey
) {
  const existingKey =
    subscription.options?.applicationServerKey;

  // Some browsers do not expose the existing key.

  if (!existingKey) return true;

  const actualKey = new Uint8Array(existingKey);

  return (
    actualKey.length === expectedKey.length &&
    actualKey.every(
      (byte, index) => byte === expectedKey[index]
    )
  );
}

/* -------------------- API -------------------- */

async function postJSON(url, body) {
  const response = await fetch(url, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    cache: "no-store",
    body: JSON.stringify(body),
  });

  const result = await response
    .json()
    .catch(() => ({}));

  if (!response.ok) {
    throw new Error(
      result.error ||
      `Request failed (${response.status})`
    );
  }

  return result;
}

/* -------------------- PREFERENCE -------------------- */

async function getPushPreference(userId) {
  if (!userId) {
    throw new Error("User ID is required");
  }

  const { data, error } = await createClient()
    .from("profiles")
    .select("notifications_enabled")
    .eq("id", userId)
    .single();

  if (error) throw error;

  return data?.notifications_enabled === true;
}

async function disablePushPreference(userId) {
  if (!userId) {
    throw new Error("User ID is required");
  }

  await postJSON("/api/push/unsubscribe", {
    endpoint: null,
    rememberPreference: true,
  });
}

/* -------------------- SERVICE WORKER -------------------- */

function getRegistration() {
  if (registrationRequest) {
    return registrationRequest;
  }

  registrationRequest = (async () => {
    const serviceWorker = navigator.serviceWorker;

    const existing =
      await serviceWorker.getRegistration("/");

    const worker =
      existing?.active ||
      existing?.waiting ||
      existing?.installing;

    const expectedUrl = new URL(
      SERVICE_WORKER_URL,
      window.location.origin
    ).href;

    if (!existing || worker?.scriptURL !== expectedUrl) {
      await serviceWorker.register(SERVICE_WORKER_URL);
    }

    return serviceWorker.ready;
  })().finally(() => {
    registrationRequest = null;
  });

  return registrationRequest;
}

async function getExistingSubscription() {
  const registration =
    await navigator.serviceWorker.getRegistration("/");

  return registration
    ? registration.pushManager.getSubscription()
    : null;
}

/* -------------------- SUBSCRIPTION -------------------- */

async function getOrCreateSubscription() {
  const applicationServerKey =
    getApplicationServerKey();

  const registration = await getRegistration();

  const { pushManager } = registration;

  const existing =
    await pushManager.getSubscription();

  if (existing) {
    if (
      !matchesApplicationServerKey(
        existing,
        applicationServerKey
      )
    ) {
      throw new Error(
        "The existing push subscription uses a different VAPID key. Disable notifications and enable them again."
      );
    }

    return existing;
  }

  return pushManager.subscribe({
    userVisibleOnly: true,
    applicationServerKey,
  });
}

async function registerSubscription(
  subscription,
  rememberPreference
) {
  await postJSON("/api/push/subscribe", {
    subscription: subscription.toJSON(),
    rememberPreference,
  });

  return ENABLED_STATE;
}

/* -------------------- RESTORE -------------------- */

export async function restorePushNotifications(userId) {
  if (!isPushSupported()) {
    return DISABLED_STATE;
  }

  const preference =
    await getPushPreference(userId);

  if (!preference) {
    return DISABLED_STATE;
  }

  if (Notification.permission !== "granted") {
    return {
      preference: true,
      enabled: false,
    };
  }

  const subscription =
    await getOrCreateSubscription();

  return registerSubscription(subscription, false);
}

/* -------------------- ENABLE -------------------- */

export async function enablePushNotifications() {
  if (!isPushSupported()) {
    throw new Error(
      "Push notifications are not supported"
    );
  }

  if (Notification.permission === "denied") {
    throw new Error(
      "Notifications are blocked by your browser"
    );
  }

  const permission =
    Notification.permission === "granted"
      ? "granted"
      : await Notification.requestPermission();

  if (permission !== "granted") {
    throw new Error(
      "Notification permission was not granted"
    );
  }

  const subscription =
    await getOrCreateSubscription();

  return registerSubscription(subscription, true);
}

/* -------------------- SUBSCRIPTION REMOVAL -------------------- */

async function removeSubscription(
  subscription,
  {
    rememberPreference,
    removeOnServerFailure = false,
  }
) {
  let error = null;

  try {
    await postJSON("/api/push/unsubscribe", {
      endpoint: subscription.endpoint,
      rememberPreference,
    });
  } catch (cause) {
    if (!removeOnServerFailure) {
      throw cause;
    }

    error = cause;
  }

  let warning = null;

  try {
    const removed =
      await subscription.unsubscribe();

    if (!removed) {
      warning =
        "Browser subscription could not be removed.";
    }
  } catch {
    warning =
      "Browser subscription cleanup failed.";
  }

  return {
    error,
    warning,
  };
}

/* -------------------- DISABLE -------------------- */

export async function disablePushNotifications(userId) {
  const subscription =
    await getExistingSubscription();

  if (!subscription) {
    await disablePushPreference(userId);

    return DISABLED_STATE;
  }

  const { warning } = await removeSubscription(
    subscription,
    {
      rememberPreference: true,
    }
  );

  return {
    ...DISABLED_STATE,
    warning,
  };
}

/* -------------------- DISMISS NOTIFICATIONS -------------------- */

async function dismissNotifications() {
  const registration =
    await navigator.serviceWorker.getRegistration("/");

  if (!registration) return;

  const notifications =
    await registration.getNotifications();

  for (const notification of notifications) {
    if (
      notification.tag === "messenger" ||
      notification.tag.startsWith("chat:")
    ) {
      notification.close();
    }
  }
}

/* -------------------- LOGOUT CLEANUP -------------------- */

export async function cleanupPushForLogout() {
  if (
    typeof navigator === "undefined" ||
    !("serviceWorker" in navigator)
  ) {
    return null;
  }

  let cleanupError = null;

  try {
    const subscription =
      await getExistingSubscription();

    if (subscription) {
      const { error, warning } =
        await removeSubscription(subscription, {
          rememberPreference: false,
          removeOnServerFailure: true,
        });

      cleanupError =
        error ||
        (warning ? new Error(warning) : null);
    }
  } catch (error) {
    cleanupError = error;
  }

  try {
    await dismissNotifications();
  } catch (error) {
    cleanupError ??= error;
  }

  return cleanupError;
}

// src/store/usePushStore.js

import { create } from "zustand";

import {
  isPushSupported,
  restorePushNotifications,
  enablePushNotifications,
  disablePushNotifications,
} from "@/lib/pushClient";

const INITIAL_STATE = {
  userId: null,

  supported: false,
  initialized: false,
  checking: false,
  loading: false,

  preference: false,
  enabled: false,

  error: null,
  warning: null,

  signingOut: false,
};

let generation = 0;
let activeRequest = null;

const usePushStore = create((set, get) => {
  function runOperation(
    operation,
    {
      fallback,
      busyKey = "loading",
      initialState = {},
    }
  ) {
    const current = generation;

    let resolve;
    let reject;

    const result = new Promise((res, rej) => {
      resolve = res;
      reject = rej;
    });

    const request = result
      .then((value) => {
        if (current !== generation) {
          return false;
        }

        set({
          initialized: true,
          enabled: value.enabled === true,
          preference: value.preference === true,
          warning: value.warning || null,
        });

        return true;
      })
      .catch((error) => {
        if (current === generation) {
          set({
            error: error?.message || fallback,
          });
        }

        return false;
      })
      .finally(() => {
        if (activeRequest === request) {
          activeRequest = null;
        }

        if (current === generation) {
          set({ [busyKey]: false });
        }
      });

    activeRequest = request;

    set({
      ...initialState,
      [busyKey]: true,
      error: null,
      warning: null,
    });

    try {
      if (
        current !== generation ||
        get().signingOut
      ) {
        throw new Error(
          "Notification operation cancelled"
        );
      }

      // Preserve browser user activation.
      resolve(operation());
    } catch (error) {
      reject(error);
    }

    return request;
  }

  function canMutate() {
    const state = get();

    return (
      Boolean(state.userId) &&
      state.supported &&
      !state.checking &&
      !state.loading &&
      !state.signingOut &&
      !activeRequest
    );
  }

  function initialize(userId, { force = false } = {}) {
    const state = get();

    if (!userId || state.signingOut) {
      return Promise.resolve(false);
    }

    if (
      state.userId &&
      state.userId !== userId
    ) {
      return Promise.resolve(false);
    }

    if (activeRequest) {
      return state.userId === userId
        ? activeRequest
        : Promise.resolve(false);
    }

    const sameUser = state.userId === userId;

    if (
      sameUser &&
      state.initialized &&
      !force
    ) {
      return Promise.resolve(true);
    }

    generation++;

    const supported = isPushSupported();

    if (!supported) {
      set({
        ...INITIAL_STATE,
        userId,
        initialized: true,
      });

      return Promise.resolve(true);
    }

    return runOperation(
      () => restorePushNotifications(userId),
      {
        fallback: "Failed to restore notifications",
        busyKey: "checking",

        initialState: {
          ...(sameUser ? {} : INITIAL_STATE),
          userId,
          supported: true,
        },
      }
    );
  }

  function enable() {
    if (!canMutate()) {
      return Promise.resolve(false);
    }

    return runOperation(
      enablePushNotifications,
      {
        fallback: "Failed to enable notifications",
      }
    );
  }

  function disable() {
    if (!canMutate()) {
      return Promise.resolve(false);
    }

    const { userId } = get();

    return runOperation(
      () => disablePushNotifications(userId),
      {
        fallback: "Failed to disable notifications",
      }
    );
  }

  async function prepareForLogout() {
    set({ signingOut: true });

    if (activeRequest) {
      await activeRequest;
    }
  }

  return {
    ...INITIAL_STATE,

    initialize,
    enable,
    disable,
    prepareForLogout,

    refresh: () =>
      initialize(get().userId, { force: true }),

    clearError: () =>
      set({
        error: null,
        warning: null,
      }),

    cancelLogout: () =>
      set({ signingOut: false }),

    reset: () => {
      generation++;
      set({ ...INITIAL_STATE });
    },
  };
});

export default usePushStore;

function expectCalledBefore(first, second) {
  expect(first).toHaveBeenCalled();
  expect(second).toHaveBeenCalled();

  expect(
    first.mock.invocationCallOrder[0]
  ).toBeLessThan(
    second.mock.invocationCallOrder[0]
  );
}

function deferred() {
  let resolve;
  let reject;

  const promise = new Promise((res, rej) => {
    resolve = res;
    reject = rej;
  });

  return {
    promise,
    resolve,
    reject,
  };
}

      expect(
        mockSignOut

    } finally {

  });

  test("resets application state only once", async () => {
    await Promise.all([
      logout(),
      logout(),
      logout(),
    ]);

    expectStoresReset();
    expectCachesCleared();
  });

  test("shares successful results with every caller", async () => {
    const results = await Promise.all([
      logout(),
      logout(),
      logout(),
    ]);

    expect(results[0]).toBe(results[1]);
    expect(results[1]).toBe(results[2]);

    expect(results[0]).toEqual({
      success: true,
      pushCleanupError: null,
    });
  });

  test("shares failures with every caller", async () => {
    const error = new Error(
      "Signout failed"
    );

    mockSignOut.mockResolvedValue({
      error,
    });

    const results = await Promise.allSettled([
      logout(),
      logout(),
      logout(),
    ]);

    for (const result of results) {
      expect(result.status).toBe("rejected");
      expect(result.reason).toBe(error);
    }

    expect(mockSignOut).toHaveBeenCalledTimes(1);

    expect(
      mockCancelLogout
    ).toHaveBeenCalledTimes(1);

    expectNoApplicationCleanup();
  });

  test("clears the active request after successful logout", async () => {
    const first = logout();

    await first;

    const second = logout();

    try {
      expect(second).not.toBe(first);
    } finally {
      await second;
    }

    expect(mockSignOut).toHaveBeenCalledTimes(2);
  });

  test("clears the active request after failed logout", async () => {
    mockSignOut.mockResolvedValueOnce({
      error: new Error("Signout failed"),
    });

    const first = logout();

    await expect(first).rejects.toThrow();

    const second = logout();

    try {
      expect(second).not.toBe(first);
    } finally {
      await second;
    }

    expect(mockSignOut).toHaveBeenCalledTimes(2);
  });
});

/* -------------------- REQUEST EFFICIENCY -------------------- */

describe("Logout - request efficiency", () => {
  test("uses no unsubscribe API requests without a subscription", async () => {
    await logout();

    expect(mockFetch).not.toHaveBeenCalled();

    expect(mockSignOut).toHaveBeenCalledTimes(1);
  });

  test("uses one unsubscribe request with an active subscription", async () => {
    enablePushSubscription();

    await logout();

    expect(mockFetch).toHaveBeenCalledTimes(1);

    expect(mockSignOut).toHaveBeenCalledTimes(1);
  });

  test("does not refresh notifications after successful logout", async () => {
    enablePushSubscription();

    await logout();

    expect(
      mockRefreshPush
    ).not.toHaveBeenCalled();
  });

  test("deduplicates ten simultaneous logout requests", async () => {
    enablePushSubscription();

    await Promise.all(
      Array.from(
        { length: 10 },
        () => logout()
      )
    );

    expect(mockFetch).toHaveBeenCalledTimes(1);

    expect(mockSignOut).toHaveBeenCalledTimes(1);

    expectStoresReset();
    expectCachesCleared();
  });
});
